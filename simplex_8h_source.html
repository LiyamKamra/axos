3// Bounded-variable revised dual simplex for LP.
4//
5// min c^T x s.t. A x - w = 0, lb <= x <= ub, l <= w <= u
6//
7// Variables 0..n-1 are the columns, n..n+m-1 the row slacks w (column -e_i).
8// The basis is m variables; the initial basis is all slacks. Ingredients:
9// - sparse LU of the basis with product-form updates (basis_lu.h)
10// - dual steepest-edge pricing
11// - bound-flipping (long-step) ratio test with Harris tolerances
12// - dual phase 1 by artificial bounds (free -> [-1000,1000], lower-only ->
13// [0,1], upper-only -> [-1,0], boxed/fixed -> [0,0])
14// - cost perturbation when the iteration stalls on dual degeneracy or a
15// (leaving, entering) pair recurs (cycling), and cost shifting for small
16// dual infeasibilities (boxed variables flip to the other bound instead)
17// A basis (nonbasic-at-bound flags) can be returned and passed back as a warm
18// start, which branch and bound and crossover use.
19//
20// The problem is scaled internally (Ruiz + Pock-Chambolle) and the solution
21// is verified on the original problem by evaluate_solution().

44 public:
45 // warm: optional starting basis; out: receives the final basis.
46 LpSolution
47 solve(const LpProblem &p, const SolverOptions &opt,
48 const SimplexBasis *warm = nullptr, SimplexBasis *out = nullptr,
49 const LpSolution *start = nullptr)
50 {
51 t_start_ = std::chrono::steady_clock::now();
52 opt_ = &opt;
53 orig_ = &p;
54 iter_cap_ = opt.max_iterations;
55 n_ = static_cast<int>(p.cols());
56 m_ = static_cast<int>(p.rows());
57 N_ = n_ + m_;
58 if (m_ == 0 || n_ == 0)
59 throw std::invalid_argument("DualSimplex::solve needs rows and columns");
60 std::string bad = p.validate();
61 if (!bad.empty()) throw std::invalid_argument("DualSimplex::solve: " + bad);
62
63 // ---- scaling ------------------------------------------------------
64 if (opt.scaling) {
65 sc_ = compute_scaling(p.A, opt.ruiz_iterations, opt.pock_chambolle_alpha);
66 q_ = apply_scaling(p, sc_);
67 } else {
68 sc_.row.assign(m_, 1.0);
69 sc_.col.assign(n_, 1.0);
70 q_ = p;
71 }
72 A_ = &q_.A;
73 AT_ = q_.A.transpose();
74 lb_.assign(N_, 0.0);
75 ub_.assign(N_, 0.0);
76 cost0_.assign(N_, 0.0);
77 for (int j = 0; j < n_; ++j) {
78 lb_[j] = q_.col_lb[j];
79 ub_[j] = q_.col_ub[j];
80 cost0_[j] = q_.c[j];
81 }
82 for (int i = 0; i < m_; ++i) {
83 lb_[n_ + i] = q_.row_lb[i];
84 ub_[n_ + i] = q_.row_ub[i];
85 }
86 cost_ = cost0_;
87 cost_orig_ = cost0_;
88 slack_idx_.resize(m_);
89 for (int i = 0; i < m_; ++i) slack_idx_[i] = i;
90 slack_val_.assign(m_, -1.0);
91 x_.assign(N_, 0.0);
92 d_.assign(N_, 0.0);
93 y_.assign(m_, 0.0);
94 status_.assign(N_, VarStatus::AtLower);
95 head_.assign(m_, -1);
96 pos_.assign(N_, -1);
97 dse_.assign(m_, 1.0);
98 iters_ = 0;
99 perturbed_ = false;
100 rng_.seed(12345);
101
102 // crossover: derive a starting basis from an interior point
103 SimplexBasis from_point;
104 crossover_ = false;
105 if (start && !warm && start->x.size() == static_cast<size_t>(n_) &&
106 start->y.size() == static_cast<size_t>(m_)) {
107 from_point = basis_from_point(*start);
108 warm = &from_point;
109 crossover_ = true;
110 }
111 Status st = run(warm);
112 LpSolution sol = extract(st);
113 if (out) out->status = status_;
114 return sol;
115 }

117 long iterations() const { return iters_; }

119 // ---- branch and bound / cutting planes (after a solve()) ----------------
120 // Re-solves the problem of the last solve() with new column bounds
121 // (original scale), keeping its scaling and matrices, from the basis warm
122 // (the parent's: still dual feasible after a bound change, so the dual
123 // simplex starts in phase 2). iteration_limit > 0 caps this solve (strong
124 // branching). The options object passed to solve() stays in use: its
125 // time_limit applies per call.
126 LpSolution
127 resolve(const std::vector<double> &col_lb, const std::vector<double> &col_ub,
128 const SimplexBasis *warm, SimplexBasis *out = nullptr, long iteration_limit = -1)
129 {
130 t_start_ = std::chrono::steady_clock::now();
131 for (int j = 0; j < n_; ++j) {
132 lb_[j] = col_lb[j] / sc_.col[j];
133 ub_[j] = col_ub[j] / sc_.col[j];
134 }
135 cost_ = cost0_;
136 iters_ = 0;
137 perturbed_ = false;
138 dse_bad_ = false;
139 crossover_ = false;
140 iter_cap_ = iteration_limit > 0 ? std::min<long>(iteration_limit, opt_->max_iterations)
141 : opt_->max_iterations;
142 Status st = run(warm);
143 LpSolution sol = extract(st);
144 if (out) out->status = status_;
145 return sol;
146 }
147
148 // Re-solves the problem of the last solve() with the costs c (original
149 // scale) and column bounds, from the basis warm, by the primal simplex: a
150 // basis optimal before a cost change stays primal feasible, so no phase 1
151 // is needed (feasibility pump). Falls back to the dual simplex when warm
152 // is not primal feasible. The costs stay in effect for resolve() until
153 // restore_costs().
154 LpSolution
155 reoptimize(const std::vector<double> &c, const std::vector<double> &col_lb, const std::vector<double> &col_ub,
156 const SimplexBasis *warm, SimplexBasis *out = nullptr, long iteration_limit = -1)
157 {
158 t_start_ = std::chrono::steady_clock::now();
159 for (int j = 0; j < n_; ++j) {
160 cost0_[j] = c[j] * sc_.col[j];
161 lb_[j] = col_lb[j] / sc_.col[j];
162 ub_[j] = col_ub[j] / sc_.col[j];
163 }
164 cost_ = cost0_;
165 iters_ = 0;
166 perturbed_ = false;
167 dse_bad_ = false;
168 crossover_ = false;
169 iter_cap_ = iteration_limit > 0 ? std::min<long>(iteration_limit, opt_->max_iterations)
170 : opt_->max_iterations;
171 Status st = run_primal(warm);
172 LpSolution sol = extract(st);
173 if (out) out->status = status_;
174 return sol;
175 }

179 restore_costs()

181 cost0_ = cost_orig_;
182 cost_ = cost0_;

186 int num_rows() const { return m_; }
187 // basic variable at basis position r (columns 0..n-1, row slacks n..n+m-1)
188 int basic_at(int r) const { return head_[r]; }
189 VarStatus status_of(int j) const { return status_[j]; }
190 // value of variable j in the original scale (row slack j >= n: the activity)
191 double value_of(int j) const { return x_[j] * scale_of(j); }
192 // reduced cost of column j (j < n) in the original scale
193 double reduced_cost(int j) const { return d_[j] / sc_.col[j]; }
194
195 // Row r of the tableau of the current basis in the original scale:
196 // x_k + sum_{j nonbasic} coef_j v_j = 0, k = basic_at(r),
197 // over the nonbasic variables v_j (columns, or row activities for j >= n).
198 void
199 tableau_row(int r, std::vector<int> &idx, std::vector<double> &coef) const
200 {
201 std::vector<double> rho(m_, 0.0);
202 rho[r] = 1.0;
203 bf_.btran(rho); // rho = B^-T e_r
204 idx.clear();
205 coef.clear();
206 const double sk = scale_of(head_[r]);
207 for (int j = 0; j < N_; ++j) {
208 if (status_[j] == VarStatus::Basic) continue;
209 const BasisFactor::Column col = column(j);
210 double a = 0;
211 for (int t = 0; t < col.nnz; ++t) a += rho[col.idx[t]] * col.val[t];
212 if (std::abs(a) < 1e-13) continue;
213 idx.push_back(j);
214 coef.push_back(a * sk / scale_of(j));
215 }
216 }

267 // Value of a nonbasic variable given its status and the bound arrays.
268 double
269 nb_value(int j, const std::vector<double> &LB, const std::vector<double> &UB) const
270 {
271 switch (status_[j]) {
272 case VarStatus::AtLower: return isfin(LB[j]) ? LB[j] : 0.0;
273 case VarStatus::AtUpper: return isfin(UB[j]) ? UB[j] : 0.0;
274 default: return 0.0;
275 }

278 // Crossover start: rank the variables by how far they are from their bounds
279 // relative to their reduced cost (interior in the primal, not in the dual =>
280 // likely basic) and take the m best as the initial basis; the rest go to
281 // their nearest bound. Dependent columns are repaired by refactor().
282 SimplexBasis
283 basis_from_point(const LpSolution &s) const
284 {
285 std::vector<double> xs(N_, 0.0), ys(m_), ds(N_, 0.0);
286 for (int j = 0; j < n_; ++j) xs[j] = s.x[j] / sc_.col[j];
287 for (int i = 0; i < m_; ++i) ys[i] = s.y[i] / sc_.row[i];
288 for (int i = 0; i < m_; ++i)
289 for (int k = A_->row_ptr()[i]; k < A_->row_ptr()[i + 1]; ++k)
290 xs[n_ + i] += A_->values()[k] * xs[A_->col_ind()[k]];
291 for (int j = 0; j < n_; ++j) {
292 double v = cost0_[j];
293 for (int k = AT_.row_ptr()[j]; k < AT_.row_ptr()[j + 1]; ++k)
294 v -= AT_.values()[k] * ys[AT_.col_ind()[k]];
295 ds[j] = v;
296 }
297 for (int i = 0; i < m_; ++i) ds[n_ + i] = ys[i];
298 std::vector<double> score(N_);
299 for (int j = 0; j < N_; ++j) {
300 if (lb_[j] == ub_[j]) { score[j] = -1.0; continue; }
301 double dp = 1e30;
302 if (isfin(lb_[j])) dp = std::min(dp, xs[j] - lb_[j]);
303 if (isfin(ub_[j])) dp = std::min(dp, ub_[j] - xs[j]);
304 dp = std::max(dp, 0.0);
305 score[j] = dp >= 1e30 ? 2.0 : dp / (dp + std::abs(ds[j]) + 1e-12);
306 }
307 std::vector<int> order(N_);
308 for (int j = 0; j < N_; ++j) order[j] = j;
309 std::sort(order.begin(), order.end(),
310 [&](int a, int b) { return score[a] > score[b]; });
311 // Take the m best-ranked columns; columns that turn out dependent are
312 // replaced by the next-best ones (slacks of the unpivoted rows at the end).
313 std::vector<int> head(order.begin(), order.begin() + m_);
314 std::vector<char> in(N_, 0);
315 for (int j : head) in[j] = 1;
316 size_t next = m_;
317 {
318 BasisFactor bf;
319 std::vector<int> sing, freerow;
320 for (int round = 0; round < 40; ++round) {
321 std::vector<BasisFactor::Column> cols(m_);
322 for (int k = 0; k < m_; ++k) cols[k] = column(head[k]);
323 const int ns = bf.factor(m_, cols, sing, freerow);
324 if (ns == 0) break;
325 for (int t = 0; t < ns; ++t) {
326 in[head[sing[t]]] = 0;
327 int repl = -1;
328 while (next < order.size() && in[order[next]]) ++next;
329 if (next < order.size() && round < 39) repl = order[next++];
330 else if (!in[n_ + freerow[t]]) repl = n_ + freerow[t];
331 if (repl < 0) repl = head[sing[t]]; // give up; refactor() repairs
332 head[sing[t]] = repl;
333 in[repl] = 1;
334 }
335 }
336 }
337 SimplexBasis b;
338 b.status.assign(N_, VarStatus::AtLower);
339 for (int j = 0; j < N_; ++j) {
340 const bool fl = isfin(lb_[j]), fu = isfin(ub_[j]);
341 if (!fl && !fu) b.status[j] = VarStatus::AtZero;
342 else if (fl && fu) b.status[j] = (xs[j] - lb_[j] <= ub_[j] - xs[j]) ? VarStatus::AtLower : VarStatus::AtUpper;
343 else b.status[j] = fl ? VarStatus::AtLower : VarStatus::AtUpper;
344 }
345 for (int j : head) b.status[j] = VarStatus::Basic;
346 if (opt_->verbose) {
347 int strong = 0, weak = 0;
348 for (int j = 0; j < N_; ++j) { strong += score[j] > 0.9; weak += score[j] > 0.5; }
349 std::printf("[simplex] crossover: m %d, score>0.9: %d, score>0.5: %d\n", m_, strong, weak);
350 }
351 return b;
352 }

354 // ---- factorization ---------------------------------------------------
355 // Factor the current basis, repairing singular columns with slacks.
356 bool
357 refactor()
358 {
359 for (int attempt = 0; attempt < 5; ++attempt) {
360 std::vector<BasisFactor::Column> cols(m_);
361 for (int k = 0; k < m_; ++k) cols[k] = column(head_[k]);
362 std::vector<int> sing, freerow;
363 int ns = bf_.factor(m_, cols, sing, freerow);
364 if (ns == 0) return true;
365 for (int t = 0; t < ns; ++t) {
366 const int k = sing[t];
367 const int leaving = head_[k];
368 const int entering = n_ + freerow[t];
369 // the slack of a free row may already be basic elsewhere
370 if (pos_[entering] >= 0) continue;
371 head_[k] = entering;
372 pos_[entering] = k;
373 status_[entering] = VarStatus::Basic;
374 pos_[leaving] = -1;
375 status_[leaving] = isfin(lb_[leaving]) ? VarStatus::AtLower
376 : isfin(ub_[leaving]) ? VarStatus::AtUpper
377 : VarStatus::AtZero;
378 }
379 }
380 return false;
381 }

383 // x_B from the nonbasic values (bounds arrays chosen by the caller).
384 void
385 compute_primal(const std::vector<double> &LB, const std::vector<double> &UB)
386 {
387 std::vector<double> rhs(m_, 0.0);
388 for (int j = 0; j < N_; ++j) {
389 if (status_[j] == VarStatus::Basic) continue;
390 const double v = nb_value(j, LB, UB);
391 x_[j] = v;
392 if (v == 0.0) continue;
393 BasisFactor::Column c = column(j);
394 for (int t = 0; t < c.nnz; ++t) rhs[c.idx[t]] -= c.val[t] * v;
395 }
396 bf_.ftran(rhs);
397 for (int k = 0; k < m_; ++k) x_[head_[k]] = rhs[k];
398 }

400 void
401 compute_dual()
402 {
403 std::vector<double> cb(m_);
404 for (int k = 0; k < m_; ++k) cb[k] = cost_[head_[k]];
405 bf_.btran(cb);
406 y_ = cb;
407 for (int j = 0; j < N_; ++j) {
408 if (status_[j] == VarStatus::Basic) { d_[j] = 0.0; continue; }
409 BasisFactor::Column c = column(j);
410 double s = cost_[j];
411 for (int t = 0; t < c.nnz; ++t) s -= c.val[t] * y_[c.idx[t]];
412 d_[j] = s;
413 }
414 }

416 // ---- dual feasibility ------------------------------------------------
417 // Choose nonbasic bounds that make the basis dual feasible where the
418 // bounds allow it. Returns the number of variables that stay infeasible.
419 int
420 place_nonbasics(const std::vector<double> &LB, const std::vector<double> &UB)
421 {
422 int infeasible = 0;
423 for (int j = 0; j < N_; ++j) {
424 if (status_[j] == VarStatus::Basic) continue;
425 const double l = LB[j], u = UB[j];
426 if (l == u) { status_[j] = VarStatus::AtLower; continue; }
427 const double dj = d_[j];
428 if (dj > dtol_) {
429 if (isfin(l)) status_[j] = VarStatus::AtLower;
430 else ++infeasible;
431 } else if (dj < -dtol_) {
432 if (isfin(u)) status_[j] = VarStatus::AtUpper;
433 else ++infeasible;
434 } else { // (nearly) zero reduced cost: keep a valid bound
435 if (status_[j] == VarStatus::AtLower && isfin(l)) {}
436 else if (status_[j] == VarStatus::AtUpper && isfin(u)) {}
437 else status_[j] = isfin(l) ? VarStatus::AtLower
438 : isfin(u) ? VarStatus::AtUpper
439 : VarStatus::AtZero;
440 }
441 }
442 return infeasible;
443 }

445 // Exact dual steepest-edge weights ||e_k^T B^-1||^2 for the current basis.
446 void
447 rebuild_dse()
448 {
449 dse_rebuild_iter_ = iters_;
450 SVec e;
451 e.init(m_);
452 for (int k = 0; k < m_; ++k) {
453 e.clear();
454 e.set(k, 1.0);
455 bf_.btran(e);
456 double n2 = 0;
457 for (int i : e.idx) n2 += e.v[i] * e.v[i];
458 dse_[k] = clamp_dse(n2);
459 }
460 }

462 // rho = B^-T e_r and the pivot row arow = rho^T [A -I], built sparsely:
463 // aidx lists the (possibly) nonzero entries of arow (basic ones included).
464 void
465 pivot_row(int r, SVec &rho, std::vector<double> &arow, std::vector<int> &aidx)
466 {
467 rho.clear();
468 rho.set(r, 1.0);
469 bf_.btran(rho);
470 for (int j : aidx) arow[j] = 0.0;
471 aidx.clear();
472 // row-wise product costs the row lengths of rho's nonzeros, column-wise nnz(A)
473 size_t rowwork = 0;
474 for (int i : rho.idx) rowwork += A_->row_ptr()[i + 1] - A_->row_ptr()[i];
475 if (rowwork * 3 < static_cast<size_t>(A_->nnz())) { // loop over the rows rho touches
476 const int *rp = A_->row_ptr(), *ci = A_->col_ind();
477 const double *va = A_->values();
478 for (int i : rho.idx) {
479 const double ri = rho.v[i];
480 if (ri == 0.0) continue;
481 for (int k = rp[i]; k < rp[i + 1]; ++k) {
482 const int j = ci[k];
483 if (arow[j] == 0.0) aidx.push_back(j);
484 arow[j] += ri * va[k];
485 if (arow[j] == 0.0) arow[j] = 1e-300; // keep the pattern entry
486 }
487 }
488 } else {
489 const int *rp = AT_.row_ptr(), *ci = AT_.col_ind();
490 const double *va = AT_.values();
491 for (int j = 0; j < n_; ++j) {
492 double sj = 0;
493 for (int k = rp[j]; k < rp[j + 1]; ++k) sj += rho.v[ci[k]] * va[k];
494 if (sj != 0.0) { arow[j] = sj; aidx.push_back(j); }
495 }
496 }
497 for (int i : rho.idx)
498 if (rho.v[i] != 0.0) { arow[n_ + i] = -rho.v[i]; aidx.push_back(n_ + i); }
499 }

501 // squared primal infeasibility of the basic variable at position k
502 void
503 set_pinf(int k, const std::vector<double> &LB, const std::vector<double> &UB)
504 {
505 const int pv = head_[k];
506 const double xv = x_[pv];
507 if (xv < LB[pv] - ptol_) pinf_[k] = (LB[pv] - xv) * (LB[pv] - xv);
508 else if (xv > UB[pv] + ptol_) pinf_[k] = (xv - UB[pv]) * (xv - UB[pv]);
509 else pinf_[k] = 0.0;
510 }

512 void
513 recompute_pinf(const std::vector<double> &LB, const std::vector<double> &UB)
514 {
515 pinf_.assign(m_, 0.0);
516 for (int k = 0; k < m_; ++k) set_pinf(k, LB, UB);
517 }

519 // ---- the dual simplex loop -------------------------------------------
520 // Returns Optimal (primal feasible for LB/UB), Infeasible (dual
521 // unbounded), or a limit / numerical status.
522 Status
523 dual_loop(const std::vector<double> &LB, const std::vector<double> &UB, bool phase1)
524 {
525 const int refactor_freq = std::min(500, std::max(100, m_ / 100));
526 SVec rho, colq, tau, rhs;
527 rho.init(m_); colq.init(m_); tau.init(m_); rhs.init(m_);
528 std::vector<double> arow(N_, 0.0);
529 std::vector<int> cand, flips, aidx;
530 std::vector<std::pair<double, int>> heap;
531 long degenerate = 0;
532 bool need_refactor = true;
533 bool fresh = false;
534 int bad_iters = 0;
535 std::vector<int> chuzr_list;
536 std::vector<char> in_list(m_, 0);
537 double chuzr_cutoff = 0.0;
538 bool need_full_scan = true;
539 const size_t kList = 64;
540 // the last 64 (leaving, entering) pairs, for the cycling check
541 std::vector<uint64_t> recent(64, ~uint64_t(0));
542 double pert_scale = 5e-6;
543 auto full_scan = [&]() {
544 std::vector<std::pair<double, int>> sc;
545 for (int k = 0; k < m_; ++k) {
546 if (pinf_[k] == 0.0) continue;
547 sc.emplace_back(pinf_[k] / dse_[k], k);
548 }
549 for (int k : chuzr_list) in_list[k] = 0;
550 chuzr_list.clear();
551 chuzr_cutoff = 0.0;
552 if (sc.size() > kList) {
553 std::nth_element(sc.begin(), sc.begin() + kList, sc.end(),
554 [](const std::pair<double, int> &a, const std::pair<double, int> &b) {
555 return a.first > b.first;
556 });
557 chuzr_cutoff = sc[kList].first; // everything outside the list scores <= this
558 sc.resize(kList);
559 }
560 for (auto &e : sc) { chuzr_list.push_back(e.second); in_list[e.second] = 1; }
561 };
562
563 while (true) {
564 if (iters_ >= iter_cap_) return Status::IterationLimit;
565 if ((iters_ & 15) == 0 && elapsed() > opt_->time_limit) return Status::TimeLimit;
566 if (need_refactor || bf_.updates() >= refactor_freq || bf_.eta_heavy(2.0)) {
567 if (!refactor()) return Status::NumericalError;
568 compute_primal(LB, UB);
569 compute_dual();
570 need_refactor = false;
571 fresh = true;
572 if (dse_bad_) { dse_bad_ = false; rebuild_dse(); }
573 // small dual infeasibilities from drift: shift costs
574 for (int j = 0; j < N_; ++j) {
575 if (status_[j] == VarStatus::Basic) continue;
576 if (LB[j] == UB[j]) continue;
577 double shift = 0;
578 if (status_[j] == VarStatus::AtLower && d_[j] < 0) shift = -d_[j];
579 else if (status_[j] == VarStatus::AtUpper && d_[j] > 0) shift = -d_[j];
580 else if (status_[j] == VarStatus::AtZero && d_[j] != 0) shift = -d_[j];
581 if (shift != 0) { cost_[j] += shift; d_[j] += shift; }
582 }
583 recompute_pinf(LB, UB);
584 need_full_scan = true;
585 }
586 // ---- CHUZR: leaving row -------------------------------------
587 // Candidate list of the largest scores pinf/dse: only positions touched
588 // by the last update can beat the cutoff of the last full scan, so the
589 // O(m) scan is repeated only when the list runs dry.
590 auto score_of = [&](int k) { return pinf_[k] == 0.0 ? 0.0 : pinf_[k] / dse_[k]; };
591 int r = -1;
592 double best = 0;
593 for (int attempt = 0; attempt < 2 && r < 0; ++attempt) {
594 if (need_full_scan) {
595 full_scan();
596 need_full_scan = false;
597 }
598 size_t w = 0;
599 for (size_t t = 0; t < chuzr_list.size(); ++t) {
600 const int k = chuzr_list[t];
601 const double sc = score_of(k);
602 if (sc == 0.0) { in_list[k] = 0; continue; }
603 chuzr_list[w++] = k;
604 if (sc > best) { best = sc; r = k; }
605 }
606 chuzr_list.resize(w);
607 if (r >= 0 && (best >= chuzr_cutoff || chuzr_cutoff == 0.0)) break;
608 if (attempt == 0 && chuzr_cutoff == 0.0) break; // the list was complete
609 r = -1;
610 best = 0;
611 need_full_scan = true;
612 }
613 if (r < 0) {
614 if (!fresh) { need_refactor = true; continue; } // confirm on a clean factor
615 return Status::Optimal;
616 }
617 const int p = head_[r];
618 const bool below = x_[p] < LB[p];
619 const double delta = below ? x_[p] - LB[p] : x_[p] - UB[p];
620
621 // ---- BTRAN and pivot row ----------------------------------------
622 pivot_row(r, rho, arow, aidx);
623 // ---- candidates for entering ------------------------------------
624 cand.clear();
625 double amax = 0;
626 for (int j : aidx)
627 if (status_[j] != VarStatus::Basic) amax = std::max(amax, std::abs(arow[j]));
628 const double pivtol = 1e-7 * std::max(1.0, amax);
629 for (int j : aidx) {
630 const VarStatus st = status_[j];
631 if (st == VarStatus::Basic) continue;
632 const double at = below ? -arow[j] : arow[j];
633 if (LB[j] == UB[j]) continue;
634 if ((st == VarStatus::AtLower && at > pivtol) ||
635 (st == VarStatus::AtUpper && at < -pivtol) ||
636 (st == VarStatus::AtZero && std::abs(at) > pivtol))
637 cand.push_back(j);
638 }
639 if (cand.empty()) return Status::Infeasible;
640 auto ratio = [&](int j) {
641 const double at = below ? -arow[j] : arow[j];
642 return std::max(0.0, d_[j] / at); // >= 0 for a dual feasible j
643 };
644 // lazy ordering: min-heap on the ratio, pop only as far as needed
645 heap.clear();
646 for (int j : cand) heap.emplace_back(ratio(j), j);
647 const auto hcmp = [](const std::pair<double, int> &x,
648 const std::pair<double, int> &y) { return x.first > y.first; };
649 std::make_heap(heap.begin(), heap.end(), hcmp);
650 size_t hend = heap.size(); // heap occupies [0, hend); popped sit behind it
651 // ---- bound-flipping ratio test -------------------------------
652 double slope = std::abs(delta);
653 flips.clear();
654 bool found = false;
655 while (hend > 0) {
656 const int j = heap.front().second;
657 const double range = UB[j] - LB[j];
658 if (!isfin(range)) { found = true; break; } // non-boxed: must enter
659 const double at = std::abs(arow[j]);
660 const double ns = slope - at * range;
661 if (ns <= ptol_) { found = true; break; } // slope used up
662 slope = ns;
663 flips.push_back(j);
664 std::pop_heap(heap.begin(), heap.begin() + hend, hcmp);
665 --hend;
666 }
667 if (!found) return Status::Infeasible; // dual unbounded
668 // Harris choice among the remaining candidates heap[0, hend)
669 // The tolerance grows while the best pivot is small relative to the row:
670 // wrong-signed reduced costs this creates are shifted away below.
671 double amax_rem = 0;
672 for (size_t t = 0; t < hend; ++t) amax_rem = std::max(amax_rem, std::abs(arow[heap[t].second]));
673 int q = heap.front().second;
674 for (double tol = dtol_;; tol *= 10) {
675 double theta_max = 1e300;
676 for (size_t t = 0; t < hend; ++t) {
677 const int j = heap[t].second;
678 theta_max = std::min(theta_max, (std::abs(d_[j]) + tol) / std::abs(arow[j]));
679 }
680 double qabs = 0;
681 for (size_t t = 0; t < hend; ++t) {
682 const int j = heap[t].second;
683 const double at = std::abs(arow[j]);
684 if (heap[t].first > theta_max) continue;
685 if (at > qabs) { qabs = at; q = j; }
686 }
687 if (qabs >= 1e-2 * amax_rem || tol >= 1e-5) break;
688 }
689 const double alpha_q = arow[q];
690 const double theta_d = d_[q] / alpha_q;
691 // Boxed candidates the Harris step leaves with a wrong-signed reduced
692 // cost move to their other bound (as in HiGHS) instead of having their
693 // cost shifted: a shift that a later step undoes can cycle (two boxed
694 // columns at d = 0 alternately flipped and shifted back, neos5 + cuts).
695 for (size_t t = 0; t < hend; ++t) {
696 const int j = heap[t].second;
697 if (j == q || !isfin(UB[j] - LB[j])) continue;
698 const double nd = d_[j] - theta_d * arow[j];
699 if ((status_[j] == VarStatus::AtLower && nd < 0) || (status_[j] == VarStatus::AtUpper && nd > 0))
700 flips.push_back(j);
701 }
702
703 // ---- FTRAN of the entering column ---------------------------------
704 colq.clear();
705 {
706 BasisFactor::Column c = column(q);
707 for (int t = 0; t < c.nnz; ++t) colq.set(c.idx[t], c.val[t]);
708 }
709 bf_.ftran(colq);
710 if (std::abs(colq.v[r] - alpha_q) > 1e-6 * (1.0 + std::abs(alpha_q)) ||
711 std::abs(colq.v[r]) < 1e-11) {
712 // the pivot row and column disagree: numerical trouble
713 if (bf_.updates() == 0 && ++bad_iters > 5) return Status::NumericalError;
714 need_refactor = true;
715 continue;
716 }
717 fresh = false;
718 bad_iters = 0;
719
720 // ---- bound flips: update x_B for the moved nonbasics -----------
721 if (!flips.empty()) {
722 rhs.clear();
723 for (int j : flips) {
724 const bool to_upper = status_[j] == VarStatus::AtLower;
725 const double nv = to_upper ? UB[j] : LB[j];
726 const double dx = nv - x_[j];
727 x_[j] = nv;
728 status_[j] = to_upper ? VarStatus::AtUpper : VarStatus::AtLower;
729 BasisFactor::Column c = column(j);
730 for (int t = 0; t < c.nnz; ++t) rhs.add(c.idx[t], c.val[t] * dx);
731 }
732 bf_.ftran(rhs);
733 for (int k : rhs.idx) {
734 x_[head_[k]] -= rhs.v[k];
735 set_pinf(k, LB, UB);
736 }
737 }
738 // ---- primal step ------------------------------------------------
739 const double target = below ? LB[p] : UB[p];
740 const double theta_p = (x_[p] - target) / colq.v[r];
741 for (int k : colq.idx) {
742 if (colq.v[k] == 0.0) continue;
743 x_[head_[k]] -= theta_p * colq.v[k];
744 set_pinf(k, LB, UB);
745 }
746 x_[q] += theta_p;
747 x_[p] = target;
748
749 // ---- dual update ----------------------------------------------------
750 for (int j : aidx) {
751 if (status_[j] == VarStatus::Basic) continue;
752 d_[j] -= theta_d * arow[j];
753 }
754 d_[q] = 0.0;
755 d_[p] = -theta_d;
756 // wrong-signed reduced costs left by the Harris step: shift costs
757 for (size_t t = 0; t < hend; ++t) {
758 const int j = heap[t].second;
759 if (j == q) continue;
760 const VarStatus st = status_[j];
761 if ((st == VarStatus::AtLower && d_[j] < 0) ||
762 (st == VarStatus::AtUpper && d_[j] > 0)) {
763 cost_[j] -= d_[j];
764 d_[j] = 0.0;
765 }
766 }
767 degenerate = std::abs(theta_d) < 1e-12 ? degenerate + 1 : 0;
768
769 // ---- DSE weights ------------------------------------------------------
770 tau.clear();
771 double wr = 0;
772 for (int i : rho.idx) {
773 tau.set(i, rho.v[i]);
774 wr += rho.v[i] * rho.v[i];
775 }
776 bf_.ftran(tau);
777 // the exact weight of the leaving row is known from rho: use it (this
778 // also repairs the drifting updated weights, as in HiGHS)
779 wr = clamp_dse(wr);
780 // the updated weight was far off: the weights have lost accuracy, rebuild
781 if (dse_[r] > 1000 * wr || dse_[r] * 1000 < wr) {
782 if (iters_ - dse_rebuild_iter_ > 30) dse_bad_ = true;
783 }
784 const double ar = colq.v[r];
785 for (int k : colq.idx) {
786 if (k == r || colq.v[k] == 0.0) continue;
787 const double ratio_k = colq.v[k] / ar;
788 dse_[k] = clamp_dse(dse_[k] - 2.0 * ratio_k * tau.v[k] + ratio_k * ratio_k * wr);
789 }
790 dse_[r] = clamp_dse(wr / (ar * ar));
791
792 // ---- basis change --------------------------------------------------------
793 head_[r] = q;
794 pos_[q] = r;
795 pos_[p] = -1;
796 status_[q] = VarStatus::Basic;
797 status_[p] = (LB[p] == UB[p] || below) ? VarStatus::AtLower : VarStatus::AtUpper;
798 set_pinf(r, LB, UB);
799 { // positions whose score changed may now beat the cutoff
800 auto touch = [&](int k) {
801 if (in_list[k]) return;
802 const double sc = score_of(k);
803 if (sc > chuzr_cutoff) { chuzr_list.push_back(k); in_list[k] = 1; }
804 };
805 for (int k : colq.idx) touch(k);
806 if (!flips.empty())
807 for (int k : rhs.idx) touch(k);
808 touch(r);
809 if (chuzr_list.size() > 4 * kList) need_full_scan = true;
810 }
811 if (!bf_.update(r, colq)) need_refactor = true;
812 ++iters_;
813 if (dse_bad_) need_refactor = true;
814 if (opt_->verbose && (iters_ % 200 == 0))
815 std::printf("[simplex] it %ld |infeas| %.3e theta_d %.2e etas %d%s\n",
816 iters_, std::abs(delta), theta_d, bf_.updates(),
817 phase1 ? " (phase 1)" : "");
818
819 // ---- stall handling: perturb costs ------------------------------------
820 // on a run of degenerate steps, and on cycling: a (leaving, entering)
821 // pair recurring a third time within 64 iterations (each repeat
822 // perturbs again, ten times stronger)
823 const uint64_t key = (static_cast<uint64_t>(p) << 32) | static_cast<uint32_t>(q);
824 int seen = 0;
825 for (uint64_t k : recent) seen += (k == key);
826 recent[iters_ & 63] = key;
827 if (seen >= 2) {
828 perturb(pert_scale);
829 pert_scale = std::min(10 * pert_scale, 1e-3);
830 std::fill(recent.begin(), recent.end(), ~uint64_t(0));
831 } else if (!phase1 && !perturbed_ && degenerate > 60) {
832 perturb(5e-7);
833 }
834 }
835 }
836
837 // ---- the primal simplex loop ---------------------------------------------
838 // Starts from a primal feasible basis (x_, d_, status_ consistent) and removes
839 // dual infeasibilities: used after the dual simplex ran on shifted costs, and
840 // as a cleanup. Devex pricing, Harris ratio test with bound flipping of the
841 // entering variable.
842 Status
843 primal_loop(const std::vector<double> &LB, const std::vector<double> &UB)
844 {
845 const int refactor_freq = std::min(500, std::max(100, m_ / 100));
846 SVec rho;
847 rho.init(m_);
848 std::vector<double> colq(m_), arow(N_, 0.0), w(N_, 1.0);
849 std::vector<int> aidx;
850 bool need_refactor = true, fresh = false;
851 int bad_iters = 0;
852 while (true) {
853 if (iters_ >= iter_cap_) return Status::IterationLimit;
854 if ((iters_ & 63) == 0 && elapsed() > opt_->time_limit) return Status::TimeLimit;
855 if (need_refactor || bf_.updates() >= refactor_freq || bf_.eta_heavy(2.0)) {
856 if (!refactor()) return Status::NumericalError;
857 compute_primal(LB, UB);
858 compute_dual();
859 need_refactor = false;
860 fresh = true;
861 }
862 // ---- CHUZC: entering variable ---------------------------------
863 int q = -1;
864 double best = 0;
865 for (int j = 0; j < N_; ++j) {
866 const VarStatus st = status_[j];
867 if (st == VarStatus::Basic || LB[j] == UB[j]) continue;
868 const double dj = d_[j];
869 double viol = 0;
870 if (st == VarStatus::AtLower) { if (dj < -dtol_) viol = -dj; }
871 else if (st == VarStatus::AtUpper) { if (dj > dtol_) viol = dj; }
872 else if (std::abs(dj) > dtol_) viol = std::abs(dj);
873 if (viol == 0) continue;
874 const double score = viol * viol / w[j];
875 if (score > best) { best = score; q = j; }
876 }
877 if (q < 0) {
878 if (!fresh) { need_refactor = true; continue; }
879 return Status::Optimal;
880 }
881 const double dir = (status_[q] == VarStatus::AtUpper ||
882 (status_[q] == VarStatus::AtZero && d_[q] > 0)) ? -1.0 : 1.0;
883 std::fill(colq.begin(), colq.end(), 0.0);
884 {
885 BasisFactor::Column c = column(q);
886 for (int t = 0; t < c.nnz; ++t) colq[c.idx[t]] = c.val[t];
887 }
888 bf_.ftran(colq);
889 double amax = 0;
890 for (int k = 0; k < m_; ++k) amax = std::max(amax, std::abs(colq[k]));
891 const double pivtol = 1e-9 * std::max(1.0, amax);
892 // ---- Harris ratio test ------------------------------------------
893 // x_B(k) changes at the rate -dir * colq[k] per unit step
894 double tmax = 1e300;
895 for (int k = 0; k < m_; ++k) {
896 const double a = dir * colq[k];
897 if (std::abs(a) <= pivtol) continue;
898 const int pv = head_[k];
899 if (a > 0) { if (isfin(LB[pv])) tmax = std::min(tmax, (x_[pv] - LB[pv] + ptol_) / a); }
900 else if (isfin(UB[pv])) tmax = std::min(tmax, (UB[pv] - x_[pv] + ptol_) / -a);
901 }
902 int r = -1;
903 double bestabs = 0, thetap = 0;
904 if (tmax < 1e300) {
905 for (int k = 0; k < m_; ++k) {
906 const double a = dir * colq[k];
907 if (std::abs(a) <= pivtol) continue;
908 const int pv = head_[k];
909 double t;
910 if (a > 0) { if (!isfin(LB[pv])) continue; t = (x_[pv] - LB[pv]) / a; }
911 else { if (!isfin(UB[pv])) continue; t = (UB[pv] - x_[pv]) / -a; }
912 if (t > tmax) continue;
913 if (std::abs(a) > bestabs) { bestabs = std::abs(a); r = k; thetap = std::max(t, 0.0); }
914 }
915 }
916 const double rng = UB[q] - LB[q];
917 if (r < 0 && !isfin(rng)) return Status::Unbounded;
918 if (r < 0 || (isfin(rng) && rng <= thetap)) { // bound flip of the entering variable
919 const double th = rng;
920 for (int k = 0; k < m_; ++k)
921 if (colq[k] != 0.0) x_[head_[k]] -= dir * th * colq[k];
922 const bool to_upper = dir > 0;
923 x_[q] = to_upper ? UB[q] : LB[q];
924 status_[q] = to_upper ? VarStatus::AtUpper : VarStatus::AtLower;
925 fresh = false;
926 ++iters_;
927 continue;
928 }
929 // ---- pivot row for the dual update -----------------------------
930 pivot_row(r, rho, arow, aidx);
931 const double alpha_r = arow[q];
932 if (std::abs(alpha_r - colq[r]) > 1e-6 * (1.0 + std::abs(colq[r])) ||
933 std::abs(colq[r]) < 1e-11) {
934 if (bf_.updates() == 0 && ++bad_iters > 5) return Status::NumericalError;
935 need_refactor = true;
936 continue;
937 }
938 fresh = false;
939 bad_iters = 0;
940 const int p = head_[r];
941 const bool p_to_lower = dir * colq[r] > 0;
942 for (int k = 0; k < m_; ++k)
943 if (colq[k] != 0.0) x_[head_[k]] -= dir * thetap * colq[k];
944 x_[q] += dir * thetap;
945 x_[p] = p_to_lower ? LB[p] : UB[p];
946 const double theta_d = d_[q] / alpha_r;
947 for (int j : aidx) {
948 if (status_[j] == VarStatus::Basic) continue;
949 d_[j] -= theta_d * arow[j];
950 }
951 d_[q] = 0.0;
952 d_[p] = -theta_d;
953 // devex reference weights
954 const double wq = w[q];
955 for (int j : aidx) {
956 if (status_[j] == VarStatus::Basic || j == q) continue;
957 const double ratio = arow[j] / alpha_r;
958 w[j] = std::max(w[j], ratio * ratio * wq);
959 }
960 w[p] = std::max(wq / (alpha_r * alpha_r), 1.0);
961 if (w[p] > 1e6) std::fill(w.begin(), w.end(), 1.0);
962 head_[r] = q;
963 pos_[q] = r;
964 pos_[p] = -1;
965 status_[q] = VarStatus::Basic;
966 status_[p] = (LB[p] == UB[p] || p_to_lower) ? VarStatus::AtLower : VarStatus::AtUpper;
967 if (!bf_.update(r, colq)) need_refactor = true;
968 ++iters_;
969 if (opt_->verbose && (iters_ % 200 == 0))
970 std::printf("[simplex] it %ld primal d_q %.3e etas %d\n", iters_, d_[q], bf_.updates());
971 }
972 }

974 // Small random cost perturbation in the dual-feasible direction.
975 void
976 perturb(double scale)
977 {
978 perturbed_ = true;
979 std::uniform_real_distribution<double> u(0.5, 1.0);
980 for (int j = 0; j < N_; ++j) {
981 const VarStatus st = status_[j];
982 if (st == VarStatus::Basic || lb_[j] == ub_[j]) continue;
983 const double e = u(rng_) * scale * (1.0 + std::abs(cost0_[j]));
984 if (st == VarStatus::AtLower) { cost_[j] += e; d_[j] += e; }
985 else if (st == VarStatus::AtUpper) { cost_[j] -= e; d_[j] -= e; }
986 }
987 }

989 // ---- driver ------------------------------------------------------------
990 // The starting basis: warm when it has m basic variables (statuses that
991 // need an infinite bound repaired), else the slack basis.
992 void
993 load_basis(const SimplexBasis *warm)
994 {
995 bool have_basis = false;
996 if (warm && warm->valid() && static_cast<int>(warm->status.size()) == N_) {
997 int nb = 0;
998 for (VarStatus s : warm->status) nb += (s == VarStatus::Basic);
999 if (nb == m_) {
1000 status_ = warm->status;
1001 int k = 0;
1002 for (int j = 0; j < N_; ++j)
1003 if (status_[j] == VarStatus::Basic) { head_[k] = j; pos_[j] = k; ++k; }
1004 for (int j = 0; j < N_; ++j) {
1005 if (status_[j] == VarStatus::Basic) continue;
1006 // repair statuses that need an infinite bound
1007 if (status_[j] == VarStatus::AtLower && !isfin(lb_[j]))
1008 status_[j] = isfin(ub_[j]) ? VarStatus::AtUpper : VarStatus::AtZero;
1009 if (status_[j] == VarStatus::AtUpper && !isfin(ub_[j]))
1010 status_[j] = isfin(lb_[j]) ? VarStatus::AtLower : VarStatus::AtZero;
1011 }
1012 have_basis = true;
1013 }
1014 }
1015 if (!have_basis) {
1016 for (int j = 0; j < n_; ++j) {
1017 status_[j] = isfin(lb_[j]) ? VarStatus::AtLower
1018 : isfin(ub_[j]) ? VarStatus::AtUpper : VarStatus::AtZero;
1019 pos_[j] = -1;
1020 }
1021 for (int i = 0; i < m_; ++i) {
1022 head_[i] = n_ + i;
1023 pos_[n_ + i] = i;
1024 status_[n_ + i] = VarStatus::Basic;
1025 }
1026 }
1027 }

1029 // Primal simplex from a primal feasible warm basis (cost changes).
1030 Status
1031 run_primal(const SimplexBasis *warm)
1032 {
1033 load_basis(warm);
1034 dse_.assign(m_, 1.0);
1035 if (!refactor()) return Status::NumericalError;
1036 compute_primal(lb_, ub_);
1037 for (int k = 0; k < m_; ++k) {
1038 const int pv = head_[k];
1039 if (x_[pv] < lb_[pv] - ptol_ || x_[pv] > ub_[pv] + ptol_) return run(warm);
1040 }
1041 compute_dual();
1042 return primal_loop(lb_, ub_);
1043 }
1044
1045 Status
1046 run(const SimplexBasis *warm)
1047 {
1048 load_basis(warm);
1049 dse_.assign(m_, 1.0);
1050 if (!refactor()) return Status::NumericalError;
1051 compute_primal(lb_, ub_);
1052 compute_dual();
1053 int infeas = place_nonbasics(lb_, ub_);
1054 if (opt_->verbose && crossover_) {
1055 int pinf = 0; double mx = 0;
1056 for (int k = 0; k < m_; ++k) {
1057 const int pv = head_[k];
1058 const double v = std::max(lb_[pv] - x_[pv], x_[pv] - ub_[pv]);
1059 if (v > ptol_) { ++pinf; mx = std::max(mx, v); }
1060 }
1061 std::printf("[simplex] crossover start: %d primal infeasible basics (max %.2e), %d dual infeasible\n", pinf, mx, infeas);
1062 }
1063 if (opt_->verbose) std::printf("[simplex] m %d n %d initial dual infeasibilities %d\n", m_, n_, infeas);
1064 if (infeas > 0 && crossover_) {
1065 // near-optimal start: remove the few dual infeasibilities by shifting
1066 // costs (the primal simplex restores them at the end)
1067 for (int j = 0; j < N_; ++j) {
1068 if (status_[j] == VarStatus::Basic || lb_[j] == ub_[j]) continue;
1069 const bool bad = (d_[j] > dtol_ && !isfin(lb_[j])) ||
1070 (d_[j] < -dtol_ && !isfin(ub_[j]));
1071 if (!bad) continue;
1072 cost_[j] -= d_[j];
1073 d_[j] = 0.0;
1074 status_[j] = isfin(lb_[j]) ? VarStatus::AtLower
1075 : isfin(ub_[j]) ? VarStatus::AtUpper : VarStatus::AtZero;
1076 }
1077 infeas = 0;
1078 }
1079 if (infeas > 0) {
1080 // ---- phase 1: artificial boxed bounds --------------------------------
1081 std::vector<double> lb1(N_), ub1(N_);
1082 for (int j = 0; j < N_; ++j) {
1083 const bool lf = isfin(lb_[j]), uf = isfin(ub_[j]);
1084 if (!lf && !uf) { lb1[j] = -1000; ub1[j] = 1000; }
1085 else if (lf && !uf) { lb1[j] = 0; ub1[j] = 1; }
1086 else if (!lf && uf) { lb1[j] = -1; ub1[j] = 0; }
1087 else { lb1[j] = 0; ub1[j] = 0; }
1088 }
1089 // start from a dual feasible placement for the artificial bounds
1090 for (int j = 0; j < N_; ++j)
1091 if (status_[j] != VarStatus::Basic)
1092 status_[j] = d_[j] >= 0 ? VarStatus::AtLower : VarStatus::AtUpper;
1093 compute_primal(lb1, ub1);
1094 Status s1 = dual_loop(lb1, ub1, true);
1095 if (s1 != Status::Optimal) return s1 == Status::Infeasible ? Status::Unbounded : s1;
1096 compute_dual();
1097 if (opt_->verbose)
1098 std::printf("[simplex] phase 1 finished after %ld iterations\n", iters_);
1099 // the artificial problem is optimal: is the basis dual feasible for
1100 // the real bounds?
1101 int left = place_nonbasics(lb_, ub_);
1102 if (left > 0) return Status::Unbounded; // dual infeasible (unbounded or infeasible)
1103 }
1104 // ---- phase 2 --------------------------------------------------------------------
1105 compute_primal(lb_, ub_);
1106 Status s2 = dual_loop(lb_, ub_, false);
1107 if (s2 != Status::Optimal) return s2;
1108 // remove perturbations and shifts, then clean up with the primal simplex
1109 if (opt_->verbose) std::printf("[simplex] dual phase 2 finished after %ld iterations\n", iters_);
1110 cost_ = cost0_;
1111 compute_primal(lb_, ub_);
1112 compute_dual();
1113 Status s3 = primal_loop(lb_, ub_);
1114 if (opt_->verbose) std::printf("[simplex] primal cleanup finished, %ld total iterations\n", iters_);
1115 return s3;
1116 }

1119 extract(Status st)
1120 {
1121 if (opt_->verbose) { // consistency of the final point in the scaled space
1122 std::vector<double> ax(m_, 0.0);
1123 const auto *rp = A_->row_ptr(); const auto *ci = A_->col_ind(); const auto *va = A_->values();
1124 for (int i = 0; i < m_; ++i)
1125 for (auto t = rp[i]; t < rp[i + 1]; ++t) ax[i] += va[t] * x_[ci[t]];
1126 double eq = 0, bl = 0;
1127 for (int i = 0; i < m_; ++i) eq = std::max(eq, std::abs(ax[i] - x_[n_ + i]));
1128 int bj = -1;
1129 for (int j = 0; j < N_; ++j) {
1130 const double v = std::max(lb_[j] - x_[j], x_[j] - ub_[j]);
1131 if (v > bl) { bl = v; bj = j; }
1132 }
1133 if (bj >= 0)
1134 std::printf("[simplex] worst var %d status %d pos %d x %.6e lb %.3e ub %.3e dse %.3e\n",
1135 bj, static_cast<int>(status_[bj]), pos_[bj], x_[bj], lb_[bj], ub_[bj],
1136 pos_[bj] >= 0 ? dse_[pos_[bj]] : 0.0);
1137 std::printf("[simplex] final: max|Ax-w| %.3e max bound viol %.3e\n", eq, bl);
1138 }
1139 std::vector<double> xv(n_), yv(m_);
1140 for (int j = 0; j < n_; ++j) xv[j] = x_[j];
1141 for (int i = 0; i < m_; ++i) yv[i] = y_[i];
1142 unscale_solution(xv, yv, sc_);
1143 LpSolution s = evaluate_solution(*orig_, xv, yv);
1144 s.status = st;
1145 s.iterations = iters_;
1146 s.seconds = elapsed();
1147 return s;
1148 }
