3// Primal-dual interior-point method for convex QP (CPU): the Mehrotra
4// predictor-corrector method of the LP solver (solver/lp/ipm.h) extended to
5//
6// min 1/2 x^T Q x + c^T x s.t. A x - w = 0, lb <= x <= ub, l <= w <= u
7//
8// with z = (x, w), multipliers lam of the equalities and bound duals sl, su.
9// Q enters in three places: the dual residual (c + Q x - A^T lam - sl + su),
10// the (1,1) block of the regularized quasi-definite augmented system
11// [ -(Q + Theta_x^{-1} + rho I) A^T ]
12// K = [ A Theta_w + delta I ]
13// (its pattern is fixed, so the ordering and symbolic analysis are done
14// once, and only the diagonal changes), and the iterative refinement of each
15// solve against the unregularized K. Unlike LP, the primal and dual steps
16// share one step length: with Q, a longer dual than primal step would break
17// the linearization of the dual residual.
18//
19// This is the accurate complement of the first-order GPU methods
20// (hpr_qp.h, pdhcg.h) for small and medium problems whose KKT factorization
21// is cheap. Known weakness: the static regularization (1e-8) perturbs the
22// constraints at that level, and a few degenerate problems are that
23// sensitive (the LISWET family converges to about 25.0 instead of 36.1 with
24// 1e-8, to the right value only with 1e-12, which other problems cannot
25// take): a proximal method of multipliers with decreasing regularization, as
26// in PIQP, is the fix.
27//
28// Before the solve, fixed columns are substituted out and empty and free
29// rows dropped (the IPM needs neither); the problem is scaled like the
30// first-order methods (qp_scaling.h). The solution is mapped back and scored
31// on the original problem with evaluate_qp(): Optimal is only reported when
32// that common metric meets opt.tol (the internal tolerances are tightened
33// up to three times when the scaled residuals were met first).

56// out = a + b
57struct AddVec {
58 double *out;
59 const double *a, *b;
60 void operator()(size_t i) const { out[i] = a[i] + b[i]; }
61};
62// y -= x
63struct SubVec {
64 double *y;
65 const double *x;
66 void operator()(size_t i) const { y[i] -= x[i]; }
67};
68// sum a b
69struct Dot {
70 const double *a, *b;
71 static constexpr int K = 1;
72 void operator()(size_t i, double *acc) const { acc[0] += a[i] * b[i]; }
73};
74// KKT diagonal source: Theta^{-1} plus diag(Q) on the columns
75struct AddDiag {
76 double *out;
77 const double *hz, *qd;
78 size_t n;
79 void operator()(size_t j) const { out[j] = hz[j] + (j < n ? qd[j] : 0.0); }
80};

83// The problem the IPM solves after fixed columns, empty rows and free rows
84// are removed, and the maps back to the original indices.
85struct QpReduction {
86 QpProblem q;
87 std::vector<int> cols; // reduced column -> original column
88 std::vector<int> rows; // reduced row -> original row (-1: the dummy row)
89 std::vector<double> x0; // original-size x with the fixed values
90 bool infeasible = false;
91
92 explicit QpReduction(const QpProblem &p)
93 {
94 const size_t n = p.cols(), m = p.rows();
95 x0.assign(n, 0.0);
96 std::vector<int> cmap(n, -1);
97 for (size_t j = 0; j < n; ++j) {
98 if (p.lp.col_lb[j] == p.lp.col_ub[j]) {
99 x0[j] = p.lp.col_lb[j];
100 } else {
101 cmap[j] = static_cast<int>(cols.size());
102 cols.push_back(static_cast<int>(j));
103 }
104 }
105 const size_t nr = cols.size();
106 // c and the offset with the fixed columns substituted
107 std::vector<double> c(nr);
108 double offset = p.lp.offset;
109 for (size_t k = 0; k < nr; ++k) c[k] = p.lp.c[cols[k]];
110 for (size_t j = 0; j < n; ++j)
111 if (cmap[j] < 0) offset += p.lp.c[j] * x0[j];
112 const bool hasQ = p.Q.nnz() > 0 && p.Q.rows() == n;
113 std::vector<int32_t> qrp{0}, qci;
114 std::vector<double> qv;
115 if (hasQ) {
116 const auto *rp = p.Q.row_ptr();
117 const auto *ci = p.Q.col_ind();
118 const double *v = p.Q.values();
119 for (size_t i = 0; i < n; ++i)
120 for (auto k = rp[i]; k < rp[i + 1]; ++k)
121 if (cmap[i] < 0 && cmap[ci[k]] < 0) offset += 0.5 * x0[i] * v[k] * x0[ci[k]];
122 else if (cmap[i] >= 0 && cmap[ci[k]] < 0) c[cmap[i]] += v[k] * x0[ci[k]];
123 for (size_t k = 0; k < nr; ++k) {
124 const int i = cols[k];
125 for (auto t = rp[i]; t < rp[i + 1]; ++t)
126 if (cmap[ci[t]] >= 0) {
127 qci.push_back(cmap[ci[t]]);
128 qv.push_back(v[t]);
129 }
130 qrp.push_back(static_cast<int32_t>(qci.size()));
131 }
132 }
133 // rows: shift by the fixed columns, drop free and empty ones
134 std::vector<int32_t> arp{0}, aci;
135 std::vector<double> av, rl, ru;
136 const auto *rp = p.lp.A.row_ptr();
137 const auto *ci = p.lp.A.col_ind();
138 const double *v = p.lp.A.values();
139 for (size_t i = 0; i < m; ++i) {
140 double shift = 0;
141 size_t kept = 0;
142 for (auto k = rp[i]; k < rp[i + 1]; ++k)
143 if (cmap[ci[k]] < 0) shift += v[k] * x0[ci[k]];
144 else if (v[k] != 0) ++kept;
145 const double lo = p.lp.row_lb[i] - shift, hi = p.lp.row_ub[i] - shift;
146 if (!std::isfinite(lo) && !std::isfinite(hi)) continue;
147 if (kept == 0) {
148 const double tol = 1e-9 * (1 + std::abs(shift));
149 if (lo > tol || hi < -tol) infeasible = true;
150 continue;
151 }
152 for (auto k = rp[i]; k < rp[i + 1]; ++k)
153 if (cmap[ci[k]] >= 0 && v[k] != 0) {
154 aci.push_back(cmap[ci[k]]);
155 av.push_back(v[k]);
156 }
157 arp.push_back(static_cast<int32_t>(aci.size()));
158 rl.push_back(lo);
159 ru.push_back(hi);
160 rows.push_back(static_cast<int>(i));
161 }
162 if (rows.empty()) { // the IPM needs a row: 0 = 0
163 arp.push_back(static_cast<int32_t>(aci.size()));
164 rl.push_back(0.0);
165 ru.push_back(0.0);
166 rows.push_back(-1);
167 }
168 q.lp.A = HostMatrix(rows.size(), nr, arp, aci, av);
169 q.Q = hasQ ? HostMatrix(nr, nr, qrp, qci, qv) : HostMatrix(nr, nr, 0);
170 q.lp.c = c;
171 q.lp.offset = offset;
172 q.lp.row_lb = rl;
173 q.lp.row_ub = ru;
174 q.lp.col_lb.resize(nr);
175 q.lp.col_ub.resize(nr);
176 for (size_t k = 0; k < nr; ++k) {
177 q.lp.col_lb[k] = p.lp.col_lb[cols[k]];
178 q.lp.col_ub[k] = p.lp.col_ub[cols[k]];
179 }
180 }
181
182 void
183 restore(const std::vector<double> &xr, const std::vector<double> &yr, size_t m,
184 std::vector<double> &x, std::vector<double> &y) const
185 {
186 x = x0;
187 for (size_t k = 0; k < cols.size(); ++k) x[cols[k]] = xr[k];
188 y.assign(m, 0.0);
189 for (size_t i = 0; i < rows.size(); ++i)
190 if (rows[i] >= 0) y[rows[i]] = yr[i];
191 }
192};

194// Flush-to-zero and denormals-are-zero for the duration of a solve, in the
195// calling thread and the OpenMP workers: near the end of an interior-point
196// solve the barrier terms span 1e-300 .. 1e300, and a factorization that
197// runs into denormal numbers is many times slower on x86.
198struct DenormalGuard {
199#if defined(_M_X64) || defined(__SSE2__)
200 static void
201 set(bool on)
202 {
203 const unsigned mask = 0x8040; // MXCSR FTZ | DAZ
204#pragma omp parallel
205 _mm_setcsr(on ? (_mm_getcsr() | mask) : (_mm_getcsr() & ~mask));
206 _mm_setcsr(on ? (_mm_getcsr() | mask) : (_mm_getcsr() & ~mask));
207 }
208 DenormalGuard() { set(true); }
209 ~DenormalGuard() { set(false); }
210#endif
211};

213class QpIpm {
214 // limits on the KKT factor: past them the solve returns NotSolved at once
215 static constexpr double kMaxFactorBytes = 2e9;
216 using Store = Cpu::HostStorage<double>;
217 using Vec = tensorET<1, double, Store>;
218 using Mat = Sparse::Csr<double, int32_t, Cpu::HostStorage>;
219 using Par = Parallel<Cpu::Backend>;
220 using Ker = Sparse::Kernels<Cpu::Backend>;
221 using Ldl = Sparse::SparseLdlt<double, int32_t, Cpu::HostStorage>;
222
223 public:
224 QpSolution
225 solve(const QpProblem &p, const QpOptions &opt)
226 {
227 using namespace ipm;
228 const auto t_start = std::chrono::steady_clock::now();
229 auto elapsed = [&] {
230 return std::chrono::duration<double>(std::chrono::steady_clock::now() - t_start)
231 .count();
232 };
233 const std::string bad = p.validate();
234 if (!bad.empty()) throw std::invalid_argument("QpIpm: " + bad);
235 const DenormalGuard ftz;
236
237 // ---- reduction and scaling -------------------------------------------
238 const QpReduction red(p);
239 auto result = [&](Status st, long iters, const std::vector<double> &xr,
240 const std::vector<double> &yr) {
241 std::vector<double> x, y;
242 red.restore(xr, yr, p.rows(), x, y);
243 QpSolution s = evaluate_qp(p, x, y);
244 s.status = st;
245 s.iterations = iters;
246 s.seconds = elapsed();
247 s.method = "ipm";
248 s.device = "cpu";
249 return s;
250 };
251 if (red.infeasible)
252 return result(Status::Infeasible, 0, std::vector<double>(red.cols.size(), 0.0),
253 std::vector<double>(red.rows.size(), 0.0));
254 const size_t n = red.q.cols(), m = red.q.rows(), N = n + m;
255 if (n == 0) {
256 QpSolution s = result(Status::Optimal, 0, {}, std::vector<double>(m, 0.0));
257 if (s.kkt() > opt.tol) s.status = Status::NumericalError;
258 return s;
259 }
260 QpScaling sc;
261 QpProblem q;
262 if (opt.scaling) {
263 sc = compute_qp_scaling(red.q, opt.ruiz_iterations, opt.pock_chambolle_alpha);
264 q = apply_qp_scaling(red.q, sc);
265 } else {
266 sc.col.assign(n, 1.0);
267 sc.row.assign(m, 1.0);
268 q = red.q;
269 }
270 const bool hasQ = q.Q.nnz() > 0;
271
272 // ---- KKT pattern: x rows (Q, diagonal, A^T), then w rows (A, diagonal) --
273 HostMatrix At = q.lp.A.transpose();
274 std::vector<int32_t> krp(N + 1, 0), kci, dxp(n), dwp(m);
275 std::vector<double> kv, qd(n, 0.0);
276 kci.reserve(2 * q.lp.A.nnz() + q.Q.nnz() + N);
277 kv.reserve(kci.capacity());
278 for (size_t j = 0; j < n; ++j) {
279 dxp[j] = static_cast<int32_t>(kci.size());
280 kci.push_back(static_cast<int32_t>(j));
281 kv.push_back(0.0);
282 if (hasQ)
283 for (auto k = q.Q.row_ptr()[j]; k < q.Q.row_ptr()[j + 1]; ++k) {
284 const auto col = q.Q.col_ind()[k];
285 if (static_cast<size_t>(col) == j) {
286 qd[j] += q.Q.values()[k];
287 } else {
288 kci.push_back(col);
289 kv.push_back(-q.Q.values()[k]);
290 }
291 }
292 for (auto k = At.row_ptr()[j]; k < At.row_ptr()[j + 1]; ++k) {
293 kci.push_back(static_cast<int32_t>(n) + At.col_ind()[k]);
294 kv.push_back(At.values()[k]);
295 }
296 krp[j + 1] = static_cast<int32_t>(kci.size());
297 }
298 for (size_t i = 0; i < m; ++i) {
299 for (auto k = q.lp.A.row_ptr()[i]; k < q.lp.A.row_ptr()[i + 1]; ++k) {
300 kci.push_back(q.lp.A.col_ind()[k]);
301 kv.push_back(q.lp.A.values()[k]);
302 }
303 dwp[i] = static_cast<int32_t>(kci.size());
304 kci.push_back(static_cast<int32_t>(n + i));
305 kv.push_back(0.0);
306 krp[n + i + 1] = static_cast<int32_t>(kci.size());
307 }
308 Mat K(N, N, krp, kci, kv);
309 const Mat &A = q.lp.A;
310 const Mat &AT = At;
311 const Mat &Qm = q.Q;
312
313 // ---- starting point (as the LP IPM) ------------------------------------
314 std::vector<double> hlb(N), hub(N), hz(N), htl(N, 1.0), htu(N, 1.0), hsl(N, 0.0),
315 hsu(N, 0.0);
316 for (size_t j = 0; j < n; ++j) {
317 hlb[j] = q.lp.col_lb[j];
318 hub[j] = q.lp.col_ub[j];
319 }
320 for (size_t i = 0; i < m; ++i) {
321 hlb[n + i] = q.lp.row_lb[i];
322 hub[n + i] = q.lp.row_ub[i];
323 }
324 double cinf = 0;
325 for (double v : q.lp.c) cinf = std::max(cinf, std::abs(v));
326 const double M = 1e2, s0 = std::max(M, cinf);
327 auto interior = [&](double lo, double hi, double target) {
328 if (lo == hi) return lo;
329 const bool lf = std::isfinite(lo), uf = std::isfinite(hi);
330 if (lf && uf) {
331 const double d = std::min(M, 0.5 * (hi - lo));
332 return std::min(std::max(target, lo + d), hi - d);
333 }
334 if (lf) return std::max(target, lo + M);
335 if (uf) return std::min(target, hi - M);
336 return target;
337 };
338 for (size_t j = 0; j < n; ++j) hz[j] = interior(hlb[j], hub[j], 0.0);
339 {
340 std::vector<double> ax(m, 0.0);
341 for (size_t i = 0; i < m; ++i)
342 for (auto k = A.row_ptr()[i]; k < A.row_ptr()[i + 1]; ++k)
343 ax[i] += A.values()[k] * hz[A.col_ind()[k]];
344 for (size_t i = 0; i < m; ++i) hz[n + i] = interior(hlb[n + i], hub[n + i], ax[i]);
345 }
346 for (size_t j = 0; j < N; ++j) {
347 if (bl(hlb[j], hub[j])) {
348 htl[j] = hz[j] - hlb[j];
349 hsl[j] = s0;
350 }
351 if (bu(hlb[j], hub[j])) {
352 htu[j] = hub[j] - hz[j];
353 hsu[j] = s0;
354 }
355 }
356 auto upload = [](const std::vector<double> &h) {
357 Vec t({std::max<size_t>(h.size(), 1)}, 0.0);
358 if (!h.empty()) std::memcpy(t.data, h.data(), h.size() * sizeof(double));
359 return t;
360 };
361 Vec z = upload(hz), tl = upload(htl), tu = upload(htu), sl = upload(hsl),
362 su = upload(hsu), lbz = upload(hlb), ubz = upload(hub);
363 Vec c = upload(q.lp.c), qdv = upload(qd);
364 Vec lam({m}, 0.0), dlam({m}, 0.0), dlam_a({m}, 0.0), rp({m}, 0.0), ax({m}, 0.0),
365 thw({m}, 0.0), tmpm({m}, 0.0);
366 Vec aty({n}, 0.0), qx({n}, 0.0), ceff({n}, 0.0), tmpn({n}, 0.0), tmpq({n}, 0.0);
367 Vec rd({N}, 0.0), hzv({N}, 0.0), hzq({N}, 0.0), rcl({N}, 0.0), rcu({N}, 0.0),
368 g({N}, 0.0), dz({N}, 0.0), dsl({N}, 0.0), dsu({N}, 0.0), dz_a({N}, 0.0),
369 dsl_a({N}, 0.0), dsu_a({N}, 0.0);
370 Vec rhs({N}, 0.0), sol({N}, 0.0), res({N}, 0.0), dsol({N}, 0.0);
371 Par par;
372 Ldl ldl(Sparse::Symmetry::Symmetric, Sparse::Ordering::MinDegree);
373 bool analyzed = false;
374
375 // scaled norms for the internal relative tests
376 double qb = 0, qc = 0;
377 for (size_t i = 0; i < m; ++i) {
378 if (std::isfinite(q.lp.row_lb[i])) qb += q.lp.row_lb[i] * q.lp.row_lb[i];
379 if (std::isfinite(q.lp.row_ub[i])) qb += q.lp.row_ub[i] * q.lp.row_ub[i];
380 }
381 for (double v : q.lp.c) qc += v * v;
382 qb = std::sqrt(qb);
383 qc = std::sqrt(qc);
384
385 double rho = 1e-8, delta = 1e-8; // static regularization (raised if a pivot fails)
386 const int ref_steps = 3;
387 double t_factor = 0, t_solve = 0; // accumulated, for the verbose log
388 double tol_eff = opt.tol;
389 int tightened = 0, stall = 0;
390 const int max_it = 300;
391
392 auto finish = [&](Status st, long iters) {
393 std::vector<double> xr(n), yr(m);
394 for (size_t j = 0; j < n; ++j) xr[j] = z.data[j] * sc.col[j];
395 for (size_t i = 0; i < m; ++i) yr[i] = lam.data[i] * sc.row[i];
396 QpSolution s = result(st, iters, xr, yr);
397 if (opt.verbose)
398 std::printf("[qp-ipm] %s after %ld iterations, %.3f s: obj %.10e eta_p %.2e "
399 "eta_d %.2e gap %.2e\n",
400 to_string(s.status), iters, s.seconds, s.primal_objective, s.rel_primal,
401 s.rel_dual, s.rel_gap);
402 return s;
403 };
404
405 // K sol = r with the current factorization, refined against the
406 // unregularized K (whose (1,1) block is -(Q + Theta_x^{-1}))
407 auto solve_kkt = [&](Vec &r, Vec &out) {
408 ldl.solve(r, out);
409 for (int step = 0; step < ref_steps; ++step) {
410 Ker::spmv(A, out.data, tmpm.data, 1.0, 0.0);
411 Ker::spmv(AT, out.data + n, tmpn.data, 1.0, 0.0);
412 if (hasQ) {
413 Ker::spmv(Qm, out.data, tmpq.data, 1.0, 0.0);
414 par.for_each(n, qipm::SubVec{tmpn.data, tmpq.data});
415 }
416 par.for_each(N, RefineRes{res.data, r.data, out.data, tmpn.data, tmpm.data,
417 hzv.data, thw.data, n});
418 par.zero();
419 par.reduce(0, N, SqNorm{res.data});
420 par.reduce(1, N, SqNorm{r.data});
421 const double *s = par.fetch();
422 if (std::sqrt(s[0]) <= 1e-13 * (1 + std::sqrt(s[1]))) return;
423 ldl.solve(res, dsol);
424 par.for_each(N, Axpy{out.data, dsol.data, 1.0});
425 }
426 };
427
428 if (opt.verbose)
429 std::printf("[qp-ipm] n %zu m %zu nnz(A) %zu nnz(Q) %zu (reduced from n %zu m %zu)\n",
430 n, m, A.nnz(), Qm.nnz(), p.cols(), p.rows());
431
432 for (long it = 0; it <= max_it; ++it) {
433 // ---- residuals and objectives ------------------------------------------
434 Ker::spmv(A, z.data, ax.data, 1.0, 0.0);
435 Ker::spmv(AT, lam.data, aty.data, 1.0, 0.0);
436 if (hasQ) Ker::spmv(Qm, z.data, qx.data, 1.0, 0.0);
437 par.for_each(n, qipm::AddVec{ceff.data, c.data, qx.data});
438 par.for_each(m, ResP{rp.data, z.data + n, ax.data});
439 par.for_each(N, ResD{rd.data, ceff.data, aty.data, lam.data, sl.data, su.data,
440 lbz.data, ubz.data, n});
441 par.zero();
442 par.reduce(0, N, Stats{rd.data, z.data, c.data, lam.data, tl.data, tu.data,
443 sl.data, su.data, lbz.data, ubz.data, n});
444 par.reduce(5, m, SqNorm{rp.data});
445 par.reduce(6, n, qipm::Dot{z.data, qx.data});
446 const double *s = par.fetch();
447 const double rd2 = s[0], cx = s[1], comp = s[3], ncomp = s[4], rp2 = s[5],
448 xqx = s[6];
449 const double pobj = cx + 0.5 * xqx + q.lp.offset;
450 const double dobj = s[2] - 0.5 * xqx + q.lp.offset;
451 const double mu = ncomp > 0 ? comp / ncomp : 0.0;
452 const double rel_p = std::sqrt(rp2) / (1 + qb);
453 const double rel_d = std::sqrt(rd2) / (1 + qc);
454 const double rel_g = std::abs(pobj - dobj) / (1 + std::abs(pobj) + std::abs(dobj));
455 if (opt.verbose >= 2)
456 std::printf("[qp-ipm] it %-3ld pobj % .8e dobj % .8e rp %.1e rd %.1e gap %.1e "
457 "mu %.1e\n",
458 it, pobj, dobj, rel_p, rel_d, rel_g, mu);
459 if (!std::isfinite(pobj) || !std::isfinite(dobj) || !std::isfinite(mu))
460 return finish(Status::NumericalError, it);
461
462 if (rel_p <= tol_eff && rel_d <= tol_eff && rel_g <= tol_eff) {
463 QpSolution cand = finish(Status::Optimal, it);
464 if (cand.kkt() <= opt.tol) return cand;
465 if (tightened >= 3) {
466 cand.status = Status::NumericalError;
467 return cand;
468 }
469 tol_eff *= 0.05;
470 ++tightened;
471 }
472 if (it >= max_it) return finish(Status::IterationLimit, it);
473 if (elapsed() > opt.time_limit) return finish(Status::TimeLimit, it);
474
475 if (mu > 0)
476 par.for_each(N, Recenter{sl.data, su.data, tl.data, tu.data, lbz.data, ubz.data,
477 mu, 1e10});
478
479 // ---- factor K -----------------------------------------------------------
480 par.for_each(N, BuildTheta{hzv.data, tl.data, tu.data, sl.data, su.data, lbz.data,
481 ubz.data});
482 par.for_each(N, qipm::AddDiag{hzq.data, hzv.data, qdv.data, n});
483 bool factored = false;
484 for (int attempt = 0; attempt < 5 && !factored; ++attempt) {
485 par.for_each(N, ScatterKkt{K.values_mut(), thw.data, dxp.data(), dwp.data(),
486 hzq.data, rho, delta, n});
487 if (!analyzed) {
488 const double ta = elapsed();
489 try {
490 ldl.analyze(K);
491 } catch (const std::bad_alloc &) { // the factors do not fit in memory
492 return finish(Status::NotSolved, it);
493 }
494 analyzed = true;
495 if (opt.verbose)
496 std::printf("[qp-ipm] KKT size %zu, nnz(L) %zu, est. %.2e flops, "
497 "analyze %.0f ms\n",
498 N, ldl.factor_nnz(), ldl.factor_flops(), (elapsed() - ta) * 1000);
499 // a factorization this large is the first-order methods' case
500 if (8.0 * static_cast<double>(ldl.factor_nnz()) > kMaxFactorBytes ||
501 ldl.factor_flops() > opt.ipm_max_flops) {
502 if (opt.verbose)
503 std::printf("[qp-ipm] factor too large (%.2e bytes, %.2e flops): giving up\n",
504 8.0 * static_cast<double>(ldl.factor_nnz()), ldl.factor_flops());
505 return finish(Status::NotSolved, it);
506 }
507 }
508 const double tf0 = elapsed();
509 factored = ldl.factorize(K);
510 t_factor += elapsed() - tf0;
511 if (opt.verbose >= 3) ldl.print_profile();
512 if (it == 0 && elapsed() + (opt.ipm_project ? 25 : 5) * t_factor > opt.time_limit) {
513 if (opt.verbose)
514 std::printf("[qp-ipm] %.0f ms per factorization: would not finish in "
515 "%.1f s, giving up\n", 1e3 * t_factor, opt.time_limit);
516 return finish(Status::NotSolved, it);
517 }
518 if (!factored) {
519 rho *= 100;
520 delta *= 100;
521 }
522 }
523 if (!factored) return finish(Status::NumericalError, it);
524
525 // ---- predictor ------------------------------------------------------------
526 par.for_each(N, CompAff{rcl.data, rcu.data, tl.data, tu.data, sl.data, su.data,
527 lbz.data, ubz.data});
528 par.for_each(N, BuildG{g.data, rd.data, rcl.data, rcu.data, tl.data, tu.data,
529 lbz.data, ubz.data});
530 par.for_each(N, PackRhs{rhs.data, g.data, rp.data, thw.data, n});
531 const double ts0 = elapsed();
532 solve_kkt(rhs, sol);
533 t_solve += elapsed() - ts0;
534 par.for_each(N, Unpack{dz_a.data, dsl_a.data, dsu_a.data, dlam_a.data, sol.data,
535 g.data, thw.data, rcl.data, rcu.data, tl.data, tu.data,
536 sl.data, su.data, lbz.data, ubz.data, n});
537 par.zero();
538 par.set_slot(0, 1e300);
539 par.set_slot(1, 1e300);
540 par.reduce_min(0, N, StepP{tl.data, tu.data, dz_a.data, lbz.data, ubz.data});
541 par.reduce_min(1, N, StepD{sl.data, su.data, dsl_a.data, dsu_a.data, lbz.data,
542 ubz.data});
543 const double *sa = par.fetch();
544 double ap_aff = std::min(1.0, sa[0]), ad_aff = std::min(1.0, sa[1]);
545 if (hasQ) ap_aff = ad_aff = std::min(ap_aff, ad_aff);
546 par.zero();
547 par.reduce(0, N, MuAff{tl.data, tu.data, sl.data, su.data, dz_a.data, dsl_a.data,
548 dsu_a.data, lbz.data, ubz.data, ap_aff, ad_aff});
549 const double mu_aff = ncomp > 0 ? par.fetch()[0] / ncomp : 0.0;
550 double sigma = mu > 0 ? std::pow(mu_aff / mu, 3.0) : 0.0;
551 sigma = std::min(1.0, std::max(sigma, 0.0));
552
553 // ---- corrector ------------------------------------------------------------
554 par.for_each(N, CompCorr{rcl.data, rcu.data, tl.data, tu.data, sl.data, su.data,
555 lbz.data, ubz.data, dz_a.data, dsl_a.data, dsu_a.data,
556 sigma * mu});
557 par.for_each(N, BuildG{g.data, rd.data, rcl.data, rcu.data, tl.data, tu.data,
558 lbz.data, ubz.data});
559 par.for_each(N, PackRhs{rhs.data, g.data, rp.data, thw.data, n});
560 const double ts1 = elapsed();
561 solve_kkt(rhs, sol);
562 t_solve += elapsed() - ts1;
563 par.for_each(N, Unpack{dz.data, dsl.data, dsu.data, dlam.data, sol.data, g.data,
564 thw.data, rcl.data, rcu.data, tl.data, tu.data, sl.data,
565 su.data, lbz.data, ubz.data, n});
566 par.zero();
567 par.set_slot(0, 1e300);
568 par.set_slot(1, 1e300);
569 par.reduce_min(0, N, StepP{tl.data, tu.data, dz.data, lbz.data, ubz.data});
570 par.reduce_min(1, N, StepD{sl.data, su.data, dsl.data, dsu.data, lbz.data,
571 ubz.data});
572 const double *sb = par.fetch();
573 const double eta = 0.995;
574 double ap = std::min(1.0, eta * sb[0]), ad = std::min(1.0, eta * sb[1]);
575 if (hasQ) ap = ad = std::min(ap, ad);
576 if (opt.verbose >= 2)
577 std::printf("[qp-ipm] ap %.2e ad %.2e sigma %.2e rho %.0e (factor %.0f ms, "
578 "solves %.0f ms so far)\n",
579 ap, ad, sigma, rho, 1e3 * t_factor, 1e3 * t_solve);
580 if (!(ap > 1e-10) && !(ad > 1e-10)) {
581 if (++stall >= 3) return finish(Status::NumericalError, it);
582 } else {
583 stall = 0;
584 }
585 par.for_each(N, UpdatePrimalDual{z.data, tl.data, tu.data, sl.data, su.data,
586 dz.data, dsl.data, dsu.data, lbz.data, ubz.data,
587 ap, ad});
588 par.for_each(m, Axpy{lam.data, dlam.data, ad});
589 }
590 return finish(Status::IterationLimit, max_it);
591 }
