3// Complemented mixed-integer rounding (c-MIR) cuts from the model rows,
4// after H. Marchand, L. Wolsey, "Aggregation and mixed integer rounding to
5// solve MIPs", Operations Research 49 (2001).
6//
7// A base inequality sum_j a_j x_j <= b (a model row, or an aggregation of
8// rows that eliminates continuous columns sitting between their bounds) is
9// turned into a mixed-integer knapsack set:
10// - every continuous x_j is replaced through its closest bound, a simple
11// one or a variable bound x_j <= c y_k + d / x_j >= c y_k + d (y_k
12// integer; found in two-column rows): x_j = L + x'_j or U - x'_j with
13// x'_j >= 0; terms with positive coefficient on x' are dropped (a
14// relaxation), the others form s = sum |c'_j| x'_j >= 0;
15// - every integer x_j is complemented to its nearer bound: z_j = x_j - l_j
16// or u_j - x_j >= 0, integer;
17// giving sum_j g_j z_j - s <= beta. For a divisor delta > 0, with
18// f0 = frac(beta / delta), the MIR inequality
19// sum_j (floor(g_j/delta) + max(0, frac(g_j/delta) - f0) / (1 - f0)) z_j
20// - s / (delta (1 - f0)) <= floor(beta / delta)
21// is valid; the divisors tried are the |g_j| of the integer columns strictly
22// inside their bounds, then the best one halved up to three times. The cut is
23// mapped back to x and kept when its efficacy (violation / norm) is at least
24// 1e-4 and its coefficients are numerically sane.
25//
26// Aggregation: when a row yields no cut, the continuous column of the base
27// inequality farthest from its bounds is eliminated with another row that
28// contains it (an equality with any multiplier, an inequality on its finite
29// side), up to max_aggr times. This is what turns flow-conservation rows
30// with variable upper bounds into flow-cover-like cuts.

44class CmirSeparator {
45 public:
46 explicit CmirSeparator(const LpProblem &p) : p_(p), At_(p.A.transpose())
47 {
48 const size_t n = p.cols(), m = p.rows();
49 vlb_.assign(n, VB());
50 vub_.assign(n, VB());
51 isint_.assign(n, 0);
52 for (size_t j = 0; j < n; ++j) isint_[j] = is_int(p, j) ? 1 : 0;
53 // variable bounds from rows a x_j + e y_k {<=, >=, =} r
54 const int32_t *rp = p.A.row_ptr(), *ci = p.A.col_ind();
55 const double *va = p.A.values();
56 for (size_t i = 0; i < m; ++i) {
57 if (rp[i + 1] - rp[i] != 2) continue;
58 int t0 = rp[i], t1 = rp[i] + 1;
59 int j = ci[t0], k = ci[t1];
60 double a = va[t0], e = va[t1];
61 if (isint_[j] && !isint_[k]) {
62 std::swap(j, k);
63 std::swap(a, e);
64 }
65 if (isint_[j] || !isint_[k] || a == 0) continue;
66 // a x_j + e y_k <= u -> x_j <= (-e/a) y_k + u/a (a > 0), >= when a < 0
67 const double c = -e / a;
68 if (std::abs(p.row_ub[i]) < 1e20) {
69 const double d = p.row_ub[i] / a;
70 if (a > 0) set_vb(vub_[j], k, c, d);
71 else set_vb(vlb_[j], k, c, d);
72 }
73 if (std::abs(p.row_lb[i]) < 1e20) {
74 const double d = p.row_lb[i] / a;
75 if (a > 0) set_vb(vlb_[j], k, c, d);
76 else set_vb(vub_[j], k, c, d);
77 }
78 }
79 dense_.assign(n, 0.0);
80 g_.assign(n, 0.0);
81 cutv_.assign(n, 0.0);
82 inrow_.assign(n, 0);
83 used_.assign(m, 0);
84 }

86 // Separates c-MIR cuts at x (the LP point over the columns) for the
87 // domain lb/ub, from the model rows; appends at most max_cuts cuts with
88 // efficacy >= min_eff to out. Stops at the deadline (steady clock).
89 int
90 separate(const std::vector<double> &x, const std::vector<double> &lb, const std::vector<double> &ub,
91 std::vector<Cut> &out, int max_cuts, std::chrono::steady_clock::time_point deadline,
92 int max_aggr = 5, double min_eff = 1e-4)
93 {
94 const size_t m = p_.rows();
95 const int32_t *rp = p_.A.row_ptr(), *ci = p_.A.col_ind();
96 const double *va = p_.A.values();
97 // start rows: rows with an integer column, nearly tight at x first
98 std::vector<std::pair<double, int>> start;
99 for (size_t i = 0; i < m; ++i) {
100 bool has_int = false;
101 double act = 0;
102 for (int32_t t = rp[i]; t < rp[i + 1]; ++t) {
103 act += va[t] * x[ci[t]];
104 has_int = has_int || isint_[ci[t]];
105 }
106 if (!has_int && !(rp[i + 1] - rp[i] > 2)) continue;
107 double slack = 1e300;
108 if (std::abs(p_.row_ub[i]) < 1e20) slack = std::min(slack, p_.row_ub[i] - act);
109 if (std::abs(p_.row_lb[i]) < 1e20) slack = std::min(slack, act - p_.row_lb[i]);
110 const double rel = slack / (1.0 + std::abs(act));
111 if (rel > 0.1) continue;
112 start.emplace_back(rel + 1e-3 * (rp[i + 1] - rp[i]) / (1.0 + n_cols()), static_cast<int>(i));
113 }
114 std::sort(start.begin(), start.end());
115 int found = 0;
116 Cut cut;
117 for (const auto &sr : start) {
118 if (found >= max_cuts || std::chrono::steady_clock::now() > deadline) break;
119 const int r = sr.second;
120 for (int side = 0; side < 2 && found < max_cuts; ++side) {
121 // side 0: a x <= u; side 1: -a x <= -l
122 const double bound = side == 0 ? p_.row_ub[r] : p_.row_lb[r];
123 if (!(std::abs(bound) < 1e20)) continue;
124 const double sgn = side == 0 ? 1.0 : -1.0;
125 clear_row();
126 add_row(r, sgn);
127 rhs_ = sgn * bound;
128 used_list_.clear();
129 used_[r] = 1;
130 used_list_.push_back(r);
131 for (int aggr = 0; aggr <= max_aggr; ++aggr) {
132 if (mir(x, lb, ub, cut) && cut.efficacy >= min_eff) {
133 if (!duplicate(cut, out)) {
134 out.push_back(cut);
135 ++found;
136 }
137 break;
138 }
139 if (aggr == max_aggr || !aggregate(x, lb, ub)) break;
140 }
141 for (int u : used_list_) used_[u] = 0;
142 }
143 }
144 clear_row();
145 return found;
146 }

149 struct VB {
150 int k = -1; // integer column y_k
151 double c = 0, d = 0; // x_j <= c y_k + d (vub) or >= (vlb)

154 struct Sub {
155 int j;
156 int type; // 0 lb, 1 ub, 2 vlb, 3 vub
157 double coef; // coefficient of x' (x' >= 0)

191 void
192 add_row(int r, double mult)
193 {
194 const int32_t *rp = p_.A.row_ptr(), *ci = p_.A.col_ind();
195 const double *va = p_.A.values();
196 for (int32_t t = rp[r]; t < rp[r + 1]; ++t) {
197 const int j = ci[t];
198 if (!inrow_[j]) {
199 inrow_[j] = 1;

203 }

206 // Eliminates the continuous column of the base inequality farthest from
207 // its (variable) bounds with an unused row; false if none applies.
208 bool
209 aggregate(const std::vector<double> &x, const std::vector<double> &lb, const std::vector<double> &ub)
210 {
211 int best_j = -1;
212 double best_d = 1e-6;
213 for (int j : nz_) {
214 if (isint_[j] || std::abs(dense_[j]) < 1e-12) continue;
215 double lo = lb[j], hi = ub[j];
216 if (vlb_[j].k >= 0) lo = std::max(lo, vlb_[j].c * x[vlb_[j].k] + vlb_[j].d);
217 if (vub_[j].k >= 0) hi = std::min(hi, vub_[j].c * x[vub_[j].k] + vub_[j].d);
218 const double d = std::min(std::abs(lo) < 1e20 ? x[j] - lo : 1e20, std::abs(hi) < 1e20 ? hi - x[j] : 1e20);
219 if (d > best_d) {
220 best_d = d;
221 best_j = j;
222 }
223 }
224 if (best_j < 0) return false;
225 const double cj = dense_[best_j];
226 int best_r = -1;
227 double best_score = 1e300, best_mult = 0;
228 const int32_t *crp = At_.row_ptr(), *cri = At_.col_ind();
229 const double *cva = At_.values();
230 const int32_t *rp = p_.A.row_ptr();
231 for (int32_t t = crp[best_j]; t < crp[best_j + 1]; ++t) {
232 const int r = cri[t];
233 if (used_[r]) continue;
234 const double a = cva[t];
235 if (std::abs(a) < 1e-9) continue;
236 const double lam = -cj / a;
237 const bool eq = p_.row_lb[r] == p_.row_ub[r];
238 double side;
239 if (eq) side = p_.row_ub[r];
240 else if (lam > 0 && std::abs(p_.row_ub[r]) < 1e20) side = p_.row_ub[r];
241 else if (lam < 0 && std::abs(p_.row_lb[r]) < 1e20) side = p_.row_lb[r];
242 else continue;
243 const double score = (eq ? 0 : 1) * 1e6 + (rp[r + 1] - rp[r]);
244 if (score < best_score) {
245 best_score = score;
246 best_r = r;
247 best_mult = lam;
248 }
249 (void)side;
250 }
251 if (best_r < 0) return false;
252 const double side = (p_.row_lb[best_r] == p_.row_ub[best_r] || best_mult > 0) ? p_.row_ub[best_r]
253 : p_.row_lb[best_r];
254 add_row(best_r, best_mult);
255 rhs_ += best_mult * side;
256 dense_[best_j] = 0; // eliminated exactly
257 used_[best_r] = 1;
258 used_list_.push_back(best_r);
259 return true;
260 }

262 // c-MIR cut from the base inequality sum dense_ x <= rhs_.
263 bool
264 mir(const std::vector<double> &x, const std::vector<double> &lb, const std::vector<double> &ub, Cut &cut)
265 {
266 // ---- substitute the continuous columns, collect integer coefficients
267 double beta = rhs_;
268 for (int j : gnz_) g_[j] = 0;
269 gnz_.clear();
270 auto addg = [&](int k, double v) {
271 if (g_[k] == 0) gnz_.push_back(k);
272 g_[k] += v;
273 if (g_[k] == 0) g_[k] = 1e-300; // keep it listed
274 };
275 std::vector<Sub> subs;
276 double sstar = 0; // value of s at x
277 bool has_frac_int = false;
278 for (int j : nz_) {
279 const double a = dense_[j];
280 if (std::abs(a) < 1e-12) continue;
281 if (isint_[j]) {
282 addg(j, a);
283 continue;
284 }
285 // candidate bounds and their distances at x
286 double dist[4] = {1e300, 1e300, 1e300, 1e300};
287 if (std::abs(lb[j]) < 1e20) dist[0] = x[j] - lb[j];
288 if (std::abs(ub[j]) < 1e20) dist[1] = ub[j] - x[j];
289 if (vlb_[j].k >= 0) dist[2] = x[j] - (vlb_[j].c * x[vlb_[j].k] + vlb_[j].d);
290 if (vub_[j].k >= 0) dist[3] = vub_[j].c * x[vub_[j].k] + vub_[j].d - x[j];
291 // prefer the bound that keeps the term (negative x' coefficient)
292 // when it is about as close; variable bounds win ties
293 int type = -1;
294 double best = 1e299;
295 for (int t : {2, 3, 0, 1}) {
296 if (dist[t] >= 1e299) continue;
297 const double dd = std::max(dist[t], 0.0);
298 if (dd < best - 1e-9) {
299 best = dd;
300 type = t;
301 }
302 }
303 if (type < 0) return false; // free continuous column
304 double coef;
305 switch (type) {
306 case 0: beta -= a * lb[j]; coef = a; break;
307 case 1: beta -= a * ub[j]; coef = -a; break;
308 case 2: beta -= a * vlb_[j].d; addg(vlb_[j].k, a * vlb_[j].c); coef = a; break;
309 default: beta -= a * vub_[j].d; addg(vub_[j].k, a * vub_[j].c); coef = -a; break;
310 }
311 if (coef < 0) {
312 subs.push_back({j, type, coef});
313 sstar += -coef * std::max(0.0, dist[type]);
314 }
315 }
316 // ---- complement the integer columns
317 struct Z {
318 int j;
319 bool comp; // z = u - x (else x - l)
320 double g, zs, range;
321 };
322 std::vector<Z> zs;
323 zs.reserve(gnz_.size());
324 for (int j : gnz_) {
325 const double gj = std::abs(g_[j]) < 1e-12 ? 0.0 : g_[j];
326 if (gj == 0) continue;
327 const double l = lb[j], u = ub[j];
328 const bool lf = std::abs(l) < 1e20, uf = std::abs(u) < 1e20;
329 if (!lf && !uf) return false;
330 const bool comp = uf && (!lf || x[j] > 0.5 * (l + u));
331 Z z;
332 z.j = j;
333 z.comp = comp;
334 if (comp) {
335 beta -= gj * u;
336 z.g = -gj;
337 z.zs = u - x[j];
338 } else {
339 beta -= gj * l;
340 z.g = gj;
341 z.zs = x[j] - l;
342 }
343 z.range = (lf && uf) ? u - l : 1e300;
344 if (z.zs > 1e-6 && z.zs < z.range - 1e-6) has_frac_int = true;
345 zs.push_back(z);
346 }
347 if (zs.empty() || !has_frac_int) return false;
348 // ---- divisors
349 auto violation = [&](double delta, double &f0out) {
350 const double bd = beta / delta;
351 const double f0 = bd - std::floor(bd);
352 f0out = f0;
353 if (f0 < 0.05 || f0 > 0.95) return -1e300; // 0.999 (as SCIP) gave stronger bounds but slower trees here
354 double lhs = 0, nrm = 0;
355 for (const Z &z : zs) {
356 const double gd = z.g / delta, fl = std::floor(gd), fj = gd - fl;
357 const double pi = fl + std::max(0.0, fj - f0) / (1 - f0);
358 lhs += pi * z.zs;
359 nrm += pi * pi;
360 }
361 const double sc = 1.0 / (delta * (1 - f0));
362 lhs -= sc * sstar;
363 for (const Sub &sb : subs) nrm += (sb.coef * sc) * (sb.coef * sc);
364 return (lhs - std::floor(bd)) / std::sqrt(std::max(nrm, 1e-300));
365 };
366 std::vector<double> deltas;
367 for (const Z &z : zs)
368 if (z.zs > 1e-6 && z.zs < z.range - 1e-6 && std::abs(z.g) > 1e-6) deltas.push_back(std::abs(z.g));
369 std::sort(deltas.begin(), deltas.end());
370 deltas.erase(std::unique(deltas.begin(), deltas.end(),
371 [](double a, double b) { return std::abs(a - b) <= 1e-9 * std::max(1.0, std::abs(a)); }),
372 deltas.end());
373 if (deltas.size() > 8) deltas.erase(deltas.begin(), deltas.end() - 8);
374 deltas.push_back(1.0);
375 double best_v = 1e-6, best_delta = 0, f0;
376 for (double dl : deltas) {
377 const double v = violation(dl, f0);
378 if (v > best_v) {
379 best_v = v;
380 best_delta = dl;
381 }
382 }
383 if (best_delta == 0) return false;
384 for (int h = 0; h < 3; ++h) {
385 const double dl = best_delta / 2;
386 const double v = violation(dl, f0);
387 if (v > best_v + 1e-9) {
388 best_v = v;
389 best_delta = dl;
390 } else {
391 break;
392 }
393 }
394 // ---- the cut in x: sum pi_z z - sc s <= floor(beta/delta)
395 const double delta = best_delta;
396 const double bd = beta / delta;
397 f0 = bd - std::floor(bd);
398 const double sc = 1.0 / (delta * (1 - f0));
399 double rhs = std::floor(bd);
400 for (int j : cnz_) cutv_[j] = 0;
401 cnz_.clear();
402 auto addc = [&](int j, double v) {
403 if (cutv_[j] == 0) cnz_.push_back(j);
404 cutv_[j] += v;
405 if (cutv_[j] == 0) cutv_[j] = 1e-300;
406 };
407 for (const Z &z : zs) {
408 const double gd = z.g / delta, fl = std::floor(gd), fj = gd - fl;
409 const double pi = fl + std::max(0.0, fj - f0) / (1 - f0);
410 if (pi == 0) continue;
411 if (z.comp) { // z = u - x
412 addc(z.j, -pi);
413 rhs -= pi * ub[z.j];
414 } else { // z = x - l
415 addc(z.j, pi);
416 rhs += pi * lb[z.j];
417 }
418 }
419 for (const Sub &sb : subs) {
420 // term -sc * |coef| * x' with x' >= 0
421 const double k = -sc * (-sb.coef); // coefficient of x' in the cut (negative)
422 const int j = sb.j;
423 switch (sb.type) {
424 case 0: addc(j, k); rhs += k * lb[j]; break; // x' = x - l
425 case 1: addc(j, -k); rhs -= k * ub[j]; break; // x' = u - x
426 case 2: addc(j, k); addc(vlb_[j].k, -k * vlb_[j].c); rhs += k * vlb_[j].d; break; // x' = x - c y - d
427 default: addc(j, -k); addc(vub_[j].k, k * vub_[j].c); rhs -= k * vub_[j].d; break; // x' = c y + d - x
428 }
429 }
430 // ---- cleanup: tiny coefficients relaxed through the bounds, then checks
431 double cmax = 0;
432 for (int j : cnz_) cmax = std::max(cmax, std::abs(cutv_[j]));
433 if (cmax < 1e-9) return false;
434 double act = 0, nrm = 0, cmin = 1e300;
435 cut.idx.clear();
436 cut.coef.clear();
437 for (int j : cnz_) {
438 double v = cutv_[j];
439 if (std::abs(v) < 1e-9 * cmax) {
440 // v x_j <= ... : drop the term using the bound that keeps validity
441 if (v > 0) {
442 if (!(std::abs(lb[j]) < 1e20)) return false;
443 rhs -= v * lb[j];
444 } else if (v < 0) {
445 if (!(std::abs(ub[j]) < 1e20)) return false;
446 rhs -= v * ub[j];
447 }
448 continue;
449 }
450 cut.idx.push_back(j);
451 cut.coef.push_back(-v); // >= form
452 act += v * x[j];
453 nrm += v * v;
454 cmin = std::min(cmin, std::abs(v));
455 }
456 if (cut.idx.empty() || cmax / cmin > 1e6) return false;
457 const double viol = act - rhs;
458 nrm = std::sqrt(nrm);
459 cut.efficacy = viol / nrm;
460 cut.rhs = -rhs;
461 if (!(cut.efficacy > 0) || !std::isfinite(cut.rhs)) return false;
462 // scale to a largest coefficient of 1
463 const double s = 1.0 / cmax;
464 for (double &v : cut.coef) v *= s;
465 cut.rhs *= s;
466 return true;
467 }

469 // a cut nearly parallel to one already found
470 static bool
471 duplicate(const Cut &c, const std::vector<Cut> &out)
472 {
473 for (size_t t = out.size() > 50 ? out.size() - 50 : 0; t < out.size(); ++t) {
474 const Cut &o = out[t];
475 if (o.idx.size() != c.idx.size()) continue;
476 bool same = true;
477 for (size_t k = 0; k < c.idx.size() && same; ++k)
478 same = o.idx[k] == c.idx[k] && std::abs(o.coef[k] - c.coef[k]) <= 1e-9;
479 if (same && std::abs(o.rhs - c.rhs) <= 1e-9) return true;
480 }
481 return false;
482 }
