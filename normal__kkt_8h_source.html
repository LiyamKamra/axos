3// Normal-equations solver for the interior-point KKT system (CPU).
4//
5// The augmented system used by Ipm,
6// [ -D A^T ] [dx] [r1]
7// [ A E ] [dy] = [r2], D = Theta_x^{-1} + rho, E = Theta_w + delta,
8// is solved by eliminating dx: S dy = r2 + A D^{-1} r1 with S = A D^{-1} A^T + E
9// (symmetric positive definite), then dx = D^{-1}(A^T dy - r1). This wins when
10// A has many more columns than rows and no dense columns (transportation and
11// assignment LPs, set covering): S is only m x m and is factored with the
12// multifrontal Cholesky. It is not used with free columns (D -> rho makes S
13// badly conditioned) or when a column touches many rows (S would be dense).

26class NormalKkt {
27 using Ldl = Sparse::SparseLdlt<double, int32_t, Cpu::HostStorage>;
28 using Mat = Sparse::Csr<double, int32_t, Cpu::HostStorage>;
29 using Vec = tensorET<1, double, Cpu::HostStorage<double>>;

31 public:
32 // Eligibility: no free columns, columns not too dense, S not too large.
33 // On success the pattern of S is built and eligible() is true.
34 NormalKkt(const LpProblem &q, Sparse::Ordering ord)
35 : ldl_(Sparse::Symmetry::SPD, ord)
36 {
37 m_ = q.rows();
38 n_ = q.cols();
39 for (size_t j = 0; j < n_; ++j)
40 if (!std::isfinite(q.col_lb[j]) && !std::isfinite(q.col_ub[j])) return;
41 A_ = &q.A;
42 AT_ = q.A.transpose();
43 const int *cp = AT_.row_ptr();
44 double work = 0;
45 for (size_t j = 0; j < n_; ++j) {
46 const double c = cp[j + 1] - cp[j];
47 if (c > 60) return; // dense column
48 work += c * c;
49 }
50 if (work > 30.0 * (q.A.nnz() + m_)) return; // S would fill in too much
51 par_ = work > 2e5;
52 // pattern of S row by row (sorted, with the diagonal)
53 std::vector<int32_t> rp(m_ + 1, 0), ci;
54 std::vector<int32_t> mark(m_, -1), tmp;
55 const int *ar = A_->row_ptr(), *ac = A_->col_ind();
56 const int *tr = AT_.row_ptr(), *tc = AT_.col_ind();
57 for (size_t i = 0; i < m_; ++i) {
58 tmp.clear();
59 mark[i] = static_cast<int32_t>(i);
60 tmp.push_back(static_cast<int32_t>(i));
61 for (int k = ar[i]; k < ar[i + 1]; ++k) {
62 const int j = ac[k];
63 for (int t = tr[j]; t < tr[j + 1]; ++t) {
64 const int r = tc[t];
65 if (mark[r] != static_cast<int32_t>(i)) {
66 mark[r] = static_cast<int32_t>(i);
67 tmp.push_back(r);
68 }
69 }
70 }
71 std::sort(tmp.begin(), tmp.end());
72 ci.insert(ci.end(), tmp.begin(), tmp.end());
73 rp[i + 1] = static_cast<int32_t>(ci.size());
74 if (ci.size() > 60u * m_ + 4000000u) return; // too large
75 }
76 srp_ = rp;
77 sci_ = ci;
78 diag_.resize(m_);
79 for (size_t i = 0; i < m_; ++i)
80 for (int32_t p = rp[i]; p < rp[i + 1]; ++p)
81 if (static_cast<size_t>(ci[p]) == i) diag_[i] = p;
82 S_ = Mat(m_, m_, rp, ci, std::vector<double>(ci.size(), 0.0));
83 dinv_.assign(n_, 0.0);
84 y_ = Vec(m_);
85 rhs_ = Vec(m_);
86 ok_ = true;
87 }
88
89 bool eligible() const { return ok_; }

94 analyzed_ = true;

100 // hz: Theta_x^{-1} for the n columns; thw: Theta_w for the m rows.
101 bool factorize(const double *hz, const double *thw, double rho, double delta)
102 {
103 for (size_t j = 0; j < n_; ++j) dinv_[j] = 1.0 / (hz[j] + rho);
104 const int *ar = A_->row_ptr(), *ac = A_->col_ind();
105 const double *av = A_->values();
106 const int *tr = AT_.row_ptr(), *tc = AT_.col_ind();
107 const double *tv = AT_.values();
108 double *sv = S_.values_mut();
109#pragma omp parallel if (par_)
110 {
111 std::vector<double> acc(m_, 0.0);
112#pragma omp for schedule(dynamic, 64)
113 for (long i = 0; i < static_cast<long>(m_); ++i) {
114 for (int k = ar[i]; k < ar[i + 1]; ++k) {
115 const int j = ac[k];
116 const double f = av[k] * dinv_[j];
117 for (int t = tr[j]; t < tr[j + 1]; ++t) acc[tc[t]] += f * tv[t];
118 }
119 for (int32_t p = srp_[i]; p < srp_[i + 1]; ++p) {
120 sv[p] = acc[sci_[p]];
121 acc[sci_[p]] = 0.0;
122 }
123 sv[diag_[i]] += thw[i] + delta;
124 }
125 }
126 return ldl_.factorize(S_);
127 }
128
129 // Solves the (regularized) augmented system: rhs = [r1 (n); r2 (m)].
130 void solve(const double *r, double *out)
131 {
132 const double *r1 = r, *r2 = r + n_;
133 // y := r2 + A D^{-1} r1
134 std::vector<double> t(n_);
135 for (size_t j = 0; j < n_; ++j) t[j] = dinv_[j] * r1[j];
136 for (size_t i = 0; i < m_; ++i) {
137 double s = r2[i];
138 for (int k = A_->row_ptr()[i]; k < A_->row_ptr()[i + 1]; ++k)
139 s += A_->values()[k] * t[A_->col_ind()[k]];
140 rhs_.data[i] = s;
141 }
142 ldl_.solve(rhs_, y_);
143 double *dx = out, *dy = out + n_;
144 for (size_t i = 0; i < m_; ++i) dy[i] = y_.data[i];
145 // dx = D^{-1} (A^T dy - r1)
146 for (size_t j = 0; j < n_; ++j) {
147 double s = 0;
148 for (int k = AT_.row_ptr()[j]; k < AT_.row_ptr()[j + 1]; ++k)
149 s += AT_.values()[k] * dy[AT_.col_ind()[k]];
150 dx[j] = dinv_[j] * (s - r1[j]);
151 }
152 }

160 std::vector<double> dinv_;
161 Vec y_, rhs_;
162 size_t m_ = 0, n_ = 0;
163 bool ok_ = false, analyzed_ = false, par_ = false;
164};
