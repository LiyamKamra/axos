3// LP presolve with a postsolve stack that restores primal AND dual solutions.
4//
5// Reductions:
6// - empty rows (feasibility check, then removed); free rows
7// (both bounds infinite) are removed too
8// - empty columns (fixed at the bound the cost prefers; may prove the
9// problem dual infeasible / unbounded)
10// - fixed columns (lb == ub: substituted out of the rows and cost)
11// - singleton rows (turned into column bounds)
12// - duplicate / parallel rows (merged; their bounds intersected)
13// applied until nothing changes.
14//
15// postsolve() maps a solution of the reduced problem back to the original
16// problem: x from the stack, y for removed rows from the rules below, and
17// z = c - A^T y recomputed on the original matrix.
18// * singleton row i on column j: if the reduced problem wants x_j at a bound
19// that this row tightened (z_j > 0 at the lower bound, z_j < 0 at the upper
20// bound), the row takes the dual: y_i = z_j / a_ij, which zeroes z_j.
21// * duplicate rows: the merged row's dual goes to whichever original row
22// supplies the binding bound.

51 // Outcome of presolve itself: NotSolved (normal), Infeasible or
52 // Unbounded (proven during presolve; no solve needed).
53 Status status() const { return status_; }
54 const LpProblem &reduced() const { return reduced_; }
55 size_t removed_rows() const { return orig_.rows() - reduced_.rows(); }
56 size_t removed_cols() const { return orig_.cols() - reduced_.cols(); }
57 // reduced column -> original column (MILP: integrality follows the columns)
58 const std::vector<int> &col_map() const { return col_map_; }

60 // Build the original-space solution from a reduced-space one. Also works
61 // when the reduced problem is empty (pass an empty LpSolution with
62 // status Optimal).
63 LpSolution
64 postsolve(const LpSolution &r) const
65 {
66 const size_t m = orig_.rows(), n = orig_.cols();
67 LpSolution s;
68 s.status = r.status;
69 s.iterations = r.iterations;
70 s.seconds = r.seconds;
71 s.x.assign(n, 0.0);
72 s.y.assign(m, 0.0);
73 for (size_t j = 0; j < col_map_.size(); ++j)
74 if (j < r.x.size()) s.x[col_map_[j]] = r.x[j];
75 for (size_t i = 0; i < row_map_.size(); ++i)
76 if (i < r.y.size()) s.y[row_map_[i]] = r.y[i];
77 for (auto &op : stack_) // primal values of removed columns
78 if (op.kind == Op::EmptyCol || op.kind == Op::FixedCol)
79 s.x[op.j] = op.value;
80
81 for (auto it = stack_.rbegin(); it != stack_.rend(); ++it) {
82 const Op &op = *it;
83 if (op.kind == Op::SingletonRow) {
84 double zj = orig_.c[op.j];
85 for (int k = At_.row_ptr()[op.j]; k < At_.row_ptr()[op.j + 1]; ++k)
86 zj -= At_.values()[k] * s.y[At_.col_ind()[k]];
87 if ((zj > 0 && op.tight_lb) || (zj < 0 && op.tight_ub))
88 s.y[op.i] = zj / op.a;
89 } else if (op.kind == Op::DuplicateRow) {
90 const double y = s.y[op.i];
91 if (y > 0) {
92 if (op.lb_i >= op.impl_lb_k) { /* row i binds */ }
93 else { s.y[op.k] = y / op.alpha; s.y[op.i] = 0; }
94 } else if (y < 0) {
95 if (op.ub_i <= op.impl_ub_k) { /* row i binds */ }
96 else { s.y[op.k] = y / op.alpha; s.y[op.i] = 0; }
97 }
98 }
99 }
100
101 // z = c - A^T y on the original matrix.
102 s.z = orig_.c;
103 for (size_t i = 0; i < m; ++i)
104 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1]; ++k)
105 s.z[orig_.A.col_ind()[k]] -= orig_.A.values()[k] * s.y[i];
106 LpSolution e = evaluate_solution(orig_, s.x, s.y);
107 e.status = s.status;
108 e.iterations = s.iterations;
109 e.seconds = s.seconds;
110 return e;
111 }

114 struct Op {
115 enum Kind { EmptyRow, EmptyCol, FixedCol, SingletonRow, DuplicateRow } kind;

118 bool tight_lb = false, tight_ub = false;
119 double lb_i = 0, ub_i = 0, impl_lb_k = 0, impl_ub_k = 0;
120 };

137 static double
138 tol_scale(double v, double tol)
139 { return tol * (1.0 + std::abs(v)); }

141 void
142 remove_row(int i)
143 {
144 row_on_[i] = 0;
145 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1]; ++k) {
146 int j = orig_.A.col_ind()[k];
147 if (col_on_[j] && --col_cnt_[j] <= 1) cq_.push_back(j);
148 }
149 }

151 void

153 {
154 col_on_[j] = 0;
155 for (int k = At_.row_ptr()[j]; k < At_.row_ptr()[j + 1]; ++k) {
156 int i = At_.col_ind()[k];
157 if (row_on_[i] && --row_cnt_[i] <= 1) rq_.push_back(i);
158 }

161 void
162 do_row(int i)
163 {
164 if (!row_on_[i]) return;
165 if (opt_.empty_rows && rlb_[i] == -kInf && rub_[i] == kInf) {
166 // a free row constrains nothing
167 Op op; op.kind = Op::EmptyRow; op.i = i;
168 stack_.push_back(op);
169 remove_row(i);
170 return;
171 }
172 if (row_cnt_[i] == 0 && opt_.empty_rows) {
173 if (rlb_[i] > tol_scale(rlb_[i], opt_.feas_tol) ||
174 rub_[i] < -tol_scale(rub_[i], opt_.feas_tol)) {
175 status_ = Status::Infeasible;
176 return;
177 }
178 Op op; op.kind = Op::EmptyRow; op.i = i;
179 stack_.push_back(op);
180 row_on_[i] = 0;
181 } else if (row_cnt_[i] == 1 && opt_.singleton_rows) {
182 int j = -1;
183 double a = 0;
184 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1]; ++k)
185 if (col_on_[orig_.A.col_ind()[k]]) {
186 j = orig_.A.col_ind()[k];
187 a = orig_.A.values()[k];
188 break;
189 }
190 if (j < 0 || a == 0) return;
191 double lo = a > 0 ? rlb_[i] / a : rub_[i] / a;
192 double hi = a > 0 ? rub_[i] / a : rlb_[i] / a;
193 Op op; op.kind = Op::SingletonRow; op.i = i; op.j = j; op.a = a;
194 if (lo > clb_[j]) { clb_[j] = lo; op.tight_lb = true; }
195 if (hi < cub_[j]) { cub_[j] = hi; op.tight_ub = true; }
196 if (clb_[j] > cub_[j]) {
197 if (clb_[j] - cub_[j] > tol_scale(cub_[j], opt_.feas_tol)) {
198 status_ = Status::Infeasible;
199 return;
200 }
201 clb_[j] = cub_[j] = 0.5 * (clb_[j] + cub_[j]);
202 }
203 stack_.push_back(op);
204 remove_row(i);
205 cq_.push_back(j);
206 }
207 }
208
209 void
210 do_col(int j)
211 {
212 if (!col_on_[j]) return;
213 if (col_cnt_[j] == 0 && opt_.empty_cols) {
214 double v;
215 if (cost_[j] > 0) v = clb_[j];
216 else if (cost_[j] < 0) v = cub_[j];
217 else v = std::min(std::max(0.0, clb_[j]), cub_[j]);
218 if (std::isinf(v)) {
219 if (cost_[j] != 0) { status_ = Status::Unbounded; return; }
220 v = std::isinf(clb_[j]) ? cub_[j] : clb_[j];
221 if (std::isinf(v)) v = 0;
222 }
223 Op op; op.kind = Op::EmptyCol; op.j = j; op.value = v;
224 stack_.push_back(op);
225 offset_ += cost_[j] * v;
226 col_on_[j] = 0;
227 } else if (clb_[j] == cub_[j] && opt_.fixed_cols) {
228 double v = clb_[j];
229 Op op; op.kind = Op::FixedCol; op.j = j; op.value = v;
230 stack_.push_back(op);
231 offset_ += cost_[j] * v;
232 for (int k = At_.row_ptr()[j]; k < At_.row_ptr()[j + 1]; ++k) {
233 int i = At_.col_ind()[k];
234 if (!row_on_[i]) continue;
235 double d = At_.values()[k] * v;
236 rlb_[i] -= d;
237 rub_[i] -= d;
238 }
239 remove_col(j);
240 }

243 bool
244 merge_duplicates()
245 {
246 // Group active rows (>= 2 entries) by a hash of their normalized
247 // pattern, then verify candidates exactly.
248 std::unordered_map<size_t, std::vector<int>> groups;
249 auto norm_of = [&](int i, int &first_col, double &first_val) {
250 first_col = -1;
251 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1]; ++k)
252 if (col_on_[orig_.A.col_ind()[k]]) {
253 first_col = orig_.A.col_ind()[k];
254 first_val = orig_.A.values()[k];
255 return;
256 }
257 };
258 for (size_t i = 0; i < orig_.rows(); ++i) {
259 if (!row_on_[i] || row_cnt_[i] < 2) continue;
260 int fc; double fv = 1;
261 norm_of(static_cast<int>(i), fc, fv);
262 size_t h = 1469598103934665603ULL;
263 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1]; ++k) {
264 int j = orig_.A.col_ind()[k];
265 if (!col_on_[j]) continue;
266 double v = orig_.A.values()[k] / fv;
267 long long q = std::llround(v * 1e9);
268 h = (h ^ static_cast<size_t>(j)) * 1099511628211ULL;
269 h = (h ^ static_cast<size_t>(q)) * 1099511628211ULL;
270 }
271 groups[h].push_back(static_cast<int>(i));
272 }
273 bool changed = false;
274 for (auto &g : groups) {
275 auto &rows = g.second;
276 for (size_t a = 0; a + 1 < rows.size(); ++a) {
277 int i = rows[a];
278 if (!row_on_[i]) continue;
279 for (size_t b = a + 1; b < rows.size(); ++b) {
280 int k = rows[b];
281 if (!row_on_[k]) continue;
282 double alpha;
283 if (!parallel(i, k, alpha)) continue;
284 // a_k = alpha * a_i => a_i x in [lb_k/alpha, ub_k/alpha]
285 double ilo = alpha > 0 ? rlb_[k] / alpha : rub_[k] / alpha;
286 double ihi = alpha > 0 ? rub_[k] / alpha : rlb_[k] / alpha;
287 Op op; op.kind = Op::DuplicateRow; op.i = i; op.k = k;
288 op.alpha = alpha; op.lb_i = rlb_[i]; op.ub_i = rub_[i];
289 op.impl_lb_k = ilo; op.impl_ub_k = ihi;
290 double nlb = std::max(rlb_[i], ilo), nub = std::min(rub_[i], ihi);
291 if (nlb > nub) {
292 if (nlb - nub > tol_scale(nub, opt_.feas_tol)) {
293 status_ = Status::Infeasible;
294 return true;
295 }
296 nlb = nub = 0.5 * (nlb + nub);
297 }
298 rlb_[i] = nlb;
299 rub_[i] = nub;
300 stack_.push_back(op);
301 remove_row(k);
302 changed = true;
303 }
304 }
305 }
306 return changed;
307 }

309 bool
310 parallel(int i, int k, double &alpha)
311 {
312 const auto *rp = orig_.A.row_ptr();
313 const auto *ci = orig_.A.col_ind();
314 const double *v = orig_.A.values();
315 int p = rp[i], q = rp[k];
316 bool have = false;
317 while (true) {
318 while (p < rp[i + 1] && !col_on_[ci[p]]) ++p;
319 while (q < rp[k + 1] && !col_on_[ci[q]]) ++q;
320 if (p >= rp[i + 1] || q >= rp[k + 1]) break;
321 if (ci[p] != ci[q]) return false;
322 double r = v[q] / v[p];
323 if (!have) { alpha = r; have = true; }
324 else if (std::abs(r - alpha) > 1e-12 * std::abs(alpha)) return false;
325 ++p; ++q;
326 }
327 while (p < rp[i + 1] && !col_on_[ci[p]]) ++p;
328 while (q < rp[k + 1] && !col_on_[ci[q]]) ++q;
329 return have && p >= rp[i + 1] && q >= rp[k + 1];
330 }

332 void
333 run()
334 {
335 const size_t m = orig_.rows(), n = orig_.cols();
336 At_ = orig_.A.transpose();
337 rlb_ = orig_.row_lb; rub_ = orig_.row_ub;
338 clb_ = orig_.col_lb; cub_ = orig_.col_ub;
339 cost_ = orig_.c;
340 offset_ = orig_.offset;
341 row_on_.assign(m, 1);
342 col_on_.assign(n, 1);
343 row_cnt_.resize(m);
344 col_cnt_.resize(n);
345 for (size_t i = 0; i < m; ++i)
346 row_cnt_[i] = orig_.A.row_ptr()[i + 1] - orig_.A.row_ptr()[i];
347 for (size_t j = 0; j < n; ++j)
348 col_cnt_[j] = At_.row_ptr()[j + 1] - At_.row_ptr()[j];
349 for (size_t i = 0; i < m; ++i) rq_.push_back(static_cast<int>(i));
350 for (size_t j = 0; j < n; ++j) cq_.push_back(static_cast<int>(j));
351
352 while (status_ == Status::NotSolved) {
353 while (status_ == Status::NotSolved && (!rq_.empty() || !cq_.empty())) {
354 if (!rq_.empty()) {
355 int i = rq_.back(); rq_.pop_back();
356 do_row(i);
357 } else {
358 int j = cq_.back(); cq_.pop_back();
359 do_col(j);
360 }
361 }
362 if (status_ != Status::NotSolved) break;
363 if (!(opt_.duplicate_rows && merge_duplicates())) break;
364 // merged rows / new bounds may enable more reductions
365 for (size_t i = 0; i < m; ++i)
366 if (row_on_[i] && row_cnt_[i] <= 1) rq_.push_back(static_cast<int>(i));
367 }
368 build_reduced();
369 }

371 void
372 build_reduced()
373 {
374 const size_t m = orig_.rows(), n = orig_.cols();
375 std::vector<int> new_row(m, -1), new_col(n, -1);
376 for (size_t i = 0; i < m; ++i)
377 if (row_on_[i]) { new_row[i] = (int)row_map_.size(); row_map_.push_back((int)i); }
378 for (size_t j = 0; j < n; ++j)
379 if (col_on_[j]) { new_col[j] = (int)col_map_.size(); col_map_.push_back((int)j); }
380 Sparse::CooBuilder<double> b(row_map_.size(), col_map_.size());
381 for (size_t i = 0; i < m; ++i) {
382 if (!row_on_[i]) continue;
383 for (int k = orig_.A.row_ptr()[i]; k < orig_.A.row_ptr()[i + 1]; ++k)
384 if (col_on_[orig_.A.col_ind()[k]])
385 b.add(new_row[i], new_col[orig_.A.col_ind()[k]], orig_.A.values()[k]);
386 }
387 reduced_.name = orig_.name + "_presolved";
388 reduced_.A = b.build();
389 for (int i : row_map_) { reduced_.row_lb.push_back(rlb_[i]); reduced_.row_ub.push_back(rub_[i]); }
390 for (int j : col_map_) {
391 reduced_.c.push_back(cost_[j]);
392 reduced_.col_lb.push_back(clb_[j]);
393 reduced_.col_ub.push_back(cub_[j]);
394 }
395 reduced_.offset = offset_;
396 reduced_.maximize = orig_.maximize;
397 }
