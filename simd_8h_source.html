3// Minimal SIMD layer for the dense kernels. simd::Pack<T> holds W lanes of T
4// in one register and provides the handful of operations the kernels need.
5// The instruction set is chosen at compile time (no runtime dispatch):
6//
7// AVX-512F Pack<double>: 8 lanes, Pack<float>: 16 lanes
8// AVX2 + FMA 4 / 8 lanes (MSVC: /arch:AVX2 implies FMA)
9// SSE2 2 / 4 lanes (fmadd is mul + add)
10// otherwise W == 1, plain scalar code
11//
12// Other element types always get the scalar Pack. Under nvcc the vector
13// paths are disabled, so headers that include this compile in .cu files.
14#pragma once
15

63// Scalar fallback (W == 1). Also used for every non-float/double T.
64template <class T> struct Pack {
65 using reg = T;
66 static constexpr int W = 1;
67 static AXOS_INLINE reg load(const T *p) { return *p; }
68 static AXOS_INLINE reg loadu(const T *p) { return *p; }
69 static AXOS_INLINE void store(T *p, reg v) { *p = v; }
70 static AXOS_INLINE void storeu(T *p, reg v) { *p = v; }
71 static AXOS_INLINE void stream(T *p, reg v) { *p = v; }
72 static AXOS_INLINE reg set1(T v) { return v; }
73 static AXOS_INLINE reg zero() { return T(0); }
74 static AXOS_INLINE reg add(reg a, reg b) { return a + b; }
75 static AXOS_INLINE reg sub(reg a, reg b) { return a - b; }
76 static AXOS_INLINE reg mul(reg a, reg b) { return a * b; }
77 static AXOS_INLINE reg div(reg a, reg b) { return a / b; }
78 static AXOS_INLINE reg fmadd(reg a, reg b, reg c) { return a * b + c; }
79 static AXOS_INLINE reg sqrt(reg a) { return std::sqrt(a); }
80 static AXOS_INLINE reg abs(reg a) { return std::abs(a); }
81 static AXOS_INLINE reg neg(reg a) { return -a; }
82 static AXOS_INLINE T hsum(reg a) { return a; }
83};
